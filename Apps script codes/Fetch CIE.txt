function createCIETrackingSheet() {
  // Create and setup Tracking Sheet (Sheet 2)
  var trackingSheet = ss.getSheetByName('CIE Tracking Sheet');

  // Insert 'Tracking Sheet' if it doesn't already exist
  if (!trackingSheet) {
    trackingSheet = ss.insertSheet('CIE Tracking Sheet');
  }
  
  // Clear existing content before writing header (optional, but clean)
  trackingSheet.clearContents();

  // Updated header (added "Campus" after "Semester")
  trackingSheet.getRange(1, 1, 1, 13).setValues([[
    "Programme Short Name",
    "Course Code",
    "Course Title",
    "Course Type",
    "Semester",
    "Campus",             // ⬅️ NEW COLUMN
    "teacher_name",
    "teacher_email",
    "CIE Max",
    "Due Date",
    "Marks File Url",
    "Completed",
    "Fetched",
    "Num Students Expected",
    "Acknowledgement"
  ]]);

  var cieSheetNames = setSheetNames().cieSheetNames;

  // Loop through each Google Sheet name in cieSheetNames array
  for (var i = 0; i < cieSheetNames.length; i++) {
    var file = DriveApp.getFilesByName(cieSheetNames[i]).next();

    // Extract last 3 letters → Campus acronym (ATP, BRN, NDG, PSN)
    var campus = cieSheetNames[i].slice(-3);

    var sheet = SpreadsheetApp.open(file).getSheetByName('Sheet1');

    var lastRow = getLastRowInColumn(sheet, 'A'); 
    var data = sheet.getRange(2, 1, lastRow - 1, 10).getValues(); // Original 10 columns A–J

    // Insert Campus column after Semester (index 4)
    for (var r = 0; r < data.length; r++) {
      data[r].splice(5, 0, campus); 
    }

    // Append data to the Tracking Sheet
    trackingSheet.getRange(
      trackingSheet.getLastRow() + 1,
      1,
      data.length,
      data[0].length
    ).setValues(data);
  }

  return trackingSheet;
}

// Helper function to get the last row in a column
function getLastRowInColumn(sheet, columnLetter) {
  var maxRow = sheet.getMaxRows();
  var lastDataCell = sheet.getRange(columnLetter + maxRow).getNextDataCell(SpreadsheetApp.Direction.UP);
  return lastDataCell.getRow();
}

function trackCIECompletionStatus() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  const cieSheet = ss.getSheetByName('CIE Tracking Sheet');
  //const eseSheet = ss.getSheetByName('ESE Tracking Sheet');

  if (!cieSheet ) { //|| !eseSheet) {
    throw new Error('One or more sheets not found. Please check the sheet names.');
  }

  // Create or get the "Completion Status" sheet
  let outputSheet = ss.getSheetByName('Completion Status');
  if (!outputSheet) {
    outputSheet = ss.insertSheet('Completion Status');
  }

  // Read data
  const cieData = cieSheet.getDataRange().getValues();
  //const eseData = eseSheet.getDataRange().getValues();

  // Header indexes (assuming headers are in Row 1)
  const programmeCol = 0; // A
  const fetchedCol = 11;  // K

  // Build set of all Programme Short Names
  const programmeSet = new Set();

  for (let i = 1; i < cieData.length; i++) {
    programmeSet.add(cieData[i][programmeCol]);
  }

  //for (let i = 1; i < eseData.length; i++) {
  //  programmeSet.add(eseData[i][programmeCol]);
  //}

  const programmeList = Array.from(programmeSet).sort();

  // Build data
  const output = [
    ['Programme Short Name', 'Num CIE Expected', 'Actual CIE',  'Completion Status']
  ];

  programmeList.forEach(prog => {
    const cieExpected = cieData.filter((row, idx) => idx > 0 && row[programmeCol] === prog).length;
    const cieActual = cieData.filter((row, idx) => idx > 0 && row[programmeCol] === prog && row[fetchedCol] === 'Done').length;

    //const eseExpected = eseData.filter((row, idx) => idx > 0 && row[programmeCol] === prog).length;
    //const eseActual = eseData.filter((row, idx) => idx > 0 && row[programmeCol] === prog && row[fetchedCol] === 'Done').length;

    let status = 'Pending';
    if (cieExpected === cieActual){ // && eseExpected === eseActual) {
      status = 'Completed';
    }

    output.push([prog, cieExpected, cieActual, status]); //, eseExpected, eseActual, status]);
  });

  // Write to Completion Status sheet
  outputSheet.clear();
  outputSheet.getRange(1, 1, output.length, output[0].length).setValues(output);
}


function trackCompletionStatus1() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  const cieSheet = ss.getSheetByName('CIE Tracking Sheet');
  const eseSheet = ss.getSheetByName('ESE Tracking Sheet');

  if (!cieSheet || !eseSheet) {
    throw new Error('One or more sheets not found. Please check the sheet names.');
  }

  // Create or get the "Completion Status" sheet
  let outputSheet = ss.getSheetByName('Completion Status');
  if (!outputSheet) {
    outputSheet = ss.insertSheet('Completion Status');
  }

  // Read data
  const cieData = cieSheet.getDataRange().getValues();
  const eseData = eseSheet.getDataRange().getValues();

  // Header indexes (assuming headers are in Row 1)
  const programmeCol = 0; // A
  const fetchedCol = 10;  // K

  // Build set of all Programme Short Names
  const programmeSet = new Set();

  for (let i = 1; i < cieData.length; i++) {
    programmeSet.add(cieData[i][programmeCol]);
  }
  for (let i = 1; i < eseData.length; i++) {
    programmeSet.add(eseData[i][programmeCol]);
  }

  const programmeList = Array.from(programmeSet).sort();

  // Build data
  const output = [
    ['Programme Short Name', 'Num CIE Expected', 'Actual CIE',
     'Num ESE Expected', 'Actual ESE', 'Completion Status']
  ];

  programmeList.forEach(prog => {
    const cieExpected = cieData.filter((row, idx) => idx > 0 && row[programmeCol] === prog).length;
    const cieActual = cieData.filter((row, idx) => idx > 0 && row[programmeCol] === prog && row[fetchedCol] === 'Done').length;

    const eseExpected = eseData.filter((row, idx) => idx > 0 && row[programmeCol] === prog).length;
    const eseActual = eseData.filter((row, idx) => idx > 0 && row[programmeCol] === prog && row[fetchedCol] === 'Done').length;

    let status = 'Pending';
    if (cieExpected === cieActual && eseExpected === eseActual) {
      status = 'Completed';
    }

    output.push([prog, cieExpected, cieActual, eseExpected, eseActual, status]);
  });

  // Write to Completion Status sheet
  outputSheet.clear();
  outputSheet.getRange(1, 1, output.length, output[0].length).setValues(output);
}


function updateCIEAggregateFromMalpractice() {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const aggregateSheet = ss.getSheetByName("CIE Aggregate Sheet");
  const malpracticeSheet = ss.getSheetByName("Malpractice");

  const aggregateData = aggregateSheet.getDataRange().getValues();
  const malpracticeData = malpracticeSheet.getDataRange().getValues();

  // ---------- Aggregate column indices ----------
  const aggHeader = aggregateData[0];

  const aggRegdCol = aggHeader.indexOf("Regd. No.");
  const aggCampusCol = aggHeader.indexOf("Campus");
  const aggCourseCol = aggHeader.indexOf("Course Code");
  const aggMarksCol = aggHeader.indexOf("CIE Marks");
  const aggPercentCol = aggHeader.indexOf("CIE Percent");
  const aggPointsCol = aggHeader.indexOf("CIE Num. Points");
  const aggRemarksCol = aggHeader.indexOf("Remarks");

  // ---------- Malpractice column indices ----------
  const malHeader = malpracticeData[0];

  const malRegdCol = malHeader.indexOf("Regd. No.");
  const malCampusCol = malHeader.indexOf("Campus");
  const malCourseCol = malHeader.indexOf("Course Code");
  const malMarksCol = malHeader.indexOf("CIE Marks");
  const malPercentCol = malHeader.indexOf("CIE Percent");
  const malPointsCol = malHeader.indexOf("CIE Num. Points");

  // ---------- Build lookup ----------
  const aggregateMap = {};

  for (let i = 1; i < aggregateData.length; i++) {
    const key =
      aggregateData[i][aggRegdCol] + "|" +
      aggregateData[i][aggCampusCol] + "|" +
      aggregateData[i][aggCourseCol];

    aggregateMap[key] = i;
  }

  // Existing backgrounds
  const bgRange = aggregateSheet.getDataRange();
  const backgrounds = bgRange.getBackgrounds();

  // ---------- Update ----------
  for (let i = 1; i < malpracticeData.length; i++) {

    const key =
      malpracticeData[i][malRegdCol] + "|" +
      malpracticeData[i][malCampusCol] + "|" +
      malpracticeData[i][malCourseCol];

    if (aggregateMap.hasOwnProperty(key)) {

      const row = aggregateMap[key];

      aggregateData[row][aggMarksCol]   = malpracticeData[i][malMarksCol];
      aggregateData[row][aggPercentCol] = malpracticeData[i][malPercentCol];
      aggregateData[row][aggPointsCol]  = malpracticeData[i][malPointsCol];
      aggregateData[row][aggRemarksCol] = "Malpractice";

      backgrounds[row].fill("#ea9999");

      Logger.log(
        "Row %s | Regd. No.: %s | Course Code: %s | row made red",
        row + 1,
        aggregateData[row][aggRegdCol],
        aggregateData[row][aggCourseCol]
      );
    }
  }

  // ---------- Write back ----------
  aggregateSheet.getDataRange().setValues(aggregateData);
  bgRange.setBackgrounds(backgrounds);
}

          var regdNosStartRow = marksSheet.getRange("marksStatementRegdNoStart").getRow();

          var regdNosCol = marksSheet.getRange("marksStatementRegdNoStart").getColumn();

          var marksCol;
          try {
            marksCol = marksSheet.getRange("totalMarks").getColumn();
          
          } catch (e) {
            marksCol = marksSheet.getRange("cieTotalMarks").getColumn();
          }
          //Logger.log(marksCol);


          var percentCol = marksSheet.getRange("ciePercent").getColumn();
          var npCol = marksSheet.getRange("cieNP").getColumn();
          var readyCol = marksSheet.getRange("ready").getColumn();
          var fetchedCol = marksSheet.getRange("fetched").getColumn();

          var progNames = marksSheet.getRange(regdNosStartRow, 1, numStudents, 1).getValues();
          var semValues = marksSheet.getRange(regdNosStartRow, 2, numStudents, 1).getValues();
          var regdNos = marksSheet.getRange(regdNosStartRow, regdNosCol, numStudents, 1).getValues();
          var cieMarks = marksSheet.getRange(regdNosStartRow, marksCol, numStudents, 1).getValues();
          var ciePercent = marksSheet.getRange(regdNosStartRow, percentCol, numStudents, 1).getValues();
          var cieNP = marksSheet.getRange(regdNosStartRow, npCol, numStudents, 1).getValues();
          var ready = marksSheet.getRange(regdNosStartRow, readyCol, numStudents, 1).getValues();
          var fetched = marksSheet.getRange(regdNosStartRow, fetchedCol, numStudents, 1).getValues();

          //Logger.log(cieMarks)

          var campus = trackingSheet.getRange(trackRow, 6).getValue();

          var output = [];

          for (var i = 0; i < numStudents; i++) {

            var marksOK = cieMarks[i][0] !== "";
            var percentOK = ciePercent[i][0] !== "";
            var npOK = cieNP[i][0] !== "";
            var readyOK = String(ready[i][0]).trim().toLowerCase() === "yes";
            var fetchedBlank = String(fetched[i][0]).trim() === "";

            if (marksOK && percentOK && npOK && readyOK && fetchedBlank) {

              output.push([
                regdNos[i][0],
                campus,
                paperCode,
                cieMarks[i][0],
                ciePercent[i][0],
                cieNP[i][0],
                progNames[i][0],
                semValues[i][0]
              ]);

              //marksSheet.getRange(regdNosStartRow + i, fetchedCol).setValue("Fetched");

              var sheetRow = regdNosStartRow + i;

              // Mark as fetched
              marksSheet.getRange(sheetRow, fetchedCol).setValue("Fetched");

              // Protect E:L of this row
              var protection = marksSheet.getRange(sheetRow, 5, 1, 8).protect();
              protection.setDescription("Protected after fetching");

              // Remove all existing editors
              protection.removeEditors(protection.getEditors());

              // If the spreadsheet belongs to a domain, also disable domain editing
              if (protection.canDomainEdit()) {
                protection.setDomainEdit(false);
              }
            }
          }

          if (output.length > 0) {
            aggregateSheet
              .getRange(aggregateSheet.getLastRow() + 1, 1, output.length, output[0].length)
              .setValues(output);
          }

          // Check whether every row is fetched
          var remaining = marksSheet
              .getRange(regdNosStartRow, fetchedCol, numStudents, 1)
              .getValues()
              .filter(r => String(r[0]).trim() === "")
              .length;

          if (remaining === 0) {
            trackingSheet.getRange(trackRow, 12).setValue("Completed");
            trackingSheet.getRange(trackRow, 13).setValue("Done");
          } else {
            allDone = false;
          }

        }
        else {
          allDone = false;
        }
         // closes else of isInternship
    }     // closes if (marksFileUrl)
  }       // closes if (fetchedStatus === "")
}         // closes for (trackRow)

  if (allDone) {
    deleteAggregateCIETrigger();
  }

  Logger.log('Data aggregation complete.');
}
