function fetchAssessmentData(programFilter) {
  try {
    logEvent("DATA_FETCHER", "2", "Starting assessment data fetch for: " + programFilter, "ℹ");

    const startTime = Date.now();

    // ── Step 1: Fetch CIE marks ──────────────────────────────────────────
    logEvent("DATA_FETCHER", "2", "Fetching CIE marks...", "ℹ");
    const cieRecords = fetchCIEMarks(programFilter);
    logEvent("DATA_FETCHER", "2", "Fetched " + cieRecords.length + " CIE records", "✓");

    // ── Step 2: Fetch ESE marks ──────────────────────────────────────────
    logEvent("DATA_FETCHER", "2", "Fetching ESE marks...", "ℹ");
    const eseRecords = fetchESEMarks(programFilter);
    logEvent("DATA_FETCHER", "2", "Fetched " + eseRecords.length + " ESE records", "✓");

    // ── Step 3: Merge CIE + ESE ──────────────────────────────────────────
    logEvent("DATA_FETCHER", "2", "Merging CIE and ESE data...", "ℹ");
    const mergedArray = mergeCIEAndESEData(cieRecords, eseRecords);

    const mergedData = {};
    for (const record of mergedArray) {
      mergedData[record.key] = record;
    }
    logEvent("DATA_FETCHER", "2", "Merged records: " + Object.keys(mergedData).length, "✓");

    // ── Step 4: Load Academic Year lookup (needed by validation) ─────────
    logEvent("DATA_FETCHER", "2", "Loading Academic Year lookup...", "ℹ");
    const academicYearLookup = _loadAcademicYearLookup();
    logEvent("DATA_FETCHER", "2", "Loaded " + Object.keys(academicYearLookup).length + " student records", "✓");

    // ── Step 5: Retry previously pending students ─────────────────────────
    logEvent("DATA_FETCHER", "2", "Retrying previously pending students...", "ℹ");

    const preValidation = validateCompleteness(cieRecords, eseRecords, academicYearLookup);
    const retryResult   = retryPendingStudents(
      mergedData,
      preValidation.blockedProgrammes,
      academicYearLookup
    );
    logEvent("DATA_FETCHER", "2",
      "Retry → resolved: " + retryResult.resolved.length +
      " | still pending: " + retryResult.stillPending.length, "✓");

    // ── Step 6: Split eligible vs blocked ────────────────────────────────
      logEvent("DATA_FETCHER", "2", "Splitting eligible and blocked records...", "ℹ");

      const historicUnknownCodes = _loadUnknownPaperCodesFromSheet();
      const allUnknownPaperCodes = new Set([
        ...(preValidation.unknownPaperCodes || []),
        ...historicUnknownCodes
      ]);

      logEvent("DATA_FETCHER", "2",
        "Unknown paper codes → runtime: " + (preValidation.unknownPaperCodes ? preValidation.unknownPaperCodes.size : 0) +
        " | from sheet: " + historicUnknownCodes.size +
        " | total blocked: " + allUnknownPaperCodes.size, "ℹ");

      const { eligibleData, blockedData, notInSMFData } = splitEligibleAndBlocked(
        mergedData,
        preValidation.blockedProgrammes,
        preValidation.leftStudents,
        academicYearLookup,
        preValidation.excludedRegnNos,
        allUnknownPaperCodes,
        preValidation.excludedPaperKeys
      );

    // ── Step 7: Write _PendingResults for newly blocked students ─────────
    writePendingResults(preValidation.blockedProgrammes, blockedData, academicYearLookup);

    // ── Step 8: Park students found in fetch but absent from SMF ─────────
    logEvent("DATA_FETCHER", "2", "Parking not-in-SMF students...", "ℹ");
    _writeNotInSMFSheet(notInSMFData);
    logEvent("DATA_FETCHER", "2",
      "Not-in-SMF students parked: " + Object.keys(notInSMFData).length, "✓");

    const duration = Math.round((Date.now() - startTime) / 1000);

    logEvent("DATA_FETCHER", "2",
      "Fetch + Validation complete in " + duration + "s" +
      " | eligible: "     + Object.keys(eligibleData).length  +
      " | blocked: "      + Object.keys(blockedData).length   +
      " | not in SMF: "   + Object.keys(notInSMFData).length, "✓");

    // ── Summary log ───────────────────────────────────────────────────────
    Logger.log("\n── FETCH SUMMARY ──────────────────────────────────");
    Logger.log("CIE records fetched    : " + cieRecords.length);
    Logger.log("ESE records fetched    : " + eseRecords.length);
    Logger.log("Merged total           : " + Object.keys(mergedData).length);
    Logger.log("Left students (L)      : " + Object.keys(preValidation.leftStudents).length);
    Logger.log("Blocked programmes     : " + Object.keys(preValidation.blockedProgrammes).length);
    Logger.log("Eligible records       : " + Object.keys(eligibleData).length);
    Logger.log("Blocked records        : " + Object.keys(blockedData).length);
    Logger.log("Not in SMF (parked)    : " + Object.keys(notInSMFData).length);
    Logger.log("Unknown paper codes    : " + (preValidation.unknownPaperCodes
                                               ? preValidation.unknownPaperCodes.size : 0));
    Logger.log("Pending resolved       : " + retryResult.resolved.length);
    Logger.log("Still pending          : " + retryResult.stillPending.length);
    Logger.log("Duration               : " + duration + "s");
    Logger.log("────────────────────────────────────────────────────\n");

    return {
      success:           true,

      // Full merged set — for reference / diagnostics
      mergedData,

      // Eligible set — ONLY this goes into the pipeline
      eligibleData,

      // Blocked set — held in _PendingResults
      blockedData,

      // Students in fetch but absent from SMF — parked in _NotInSMFStudents
      notInSMFData,

      // Counts
      totalRecords:      Object.keys(mergedData).length,
      eligibleCount:     Object.keys(eligibleData).length,
      blockedCount:      Object.keys(blockedData).length,
      notInSMFCount:     Object.keys(notInSMFData).length,
      cieCount:          cieRecords.length,
      eseCount:          eseRecords.length,

      // Validation detail — available if callers need it
      validation:        preValidation,
      retryResult,

      duration
    };

  } catch (error) {
    logEvent("ERROR", "2", "Assessment data fetch failed: " + error.message, "✗");
    Logger.log("ERROR in fetchAssessmentData: " + error.message);
    Logger.log("Stack: " + error.stack);

    return {
      success:       false,
      mergedData:    {},
      eligibleData:  {},
      blockedData:   {},
      notInSMFData:  {},
      totalRecords:  0,
      eligibleCount: 0,
      blockedCount:  0,
      notInSMFCount: 0,
      error:         error.message
    };
  }
}

// SECTION 2: FETCH CIE MARKS
// ════════════════════════════════════════════════════════════════════════════
function fetchCIEMarks(programFilter) {
  try {
    Logger.log("Fetching CIE marks for program: " + programFilter);

    const cieFileId    = getCIEFileId();
    const cieSheetName = getCIESheetName();
    const cieFile      = SpreadsheetApp.openById(cieFileId);
    const cieSheet     = cieFile.getSheetByName(cieSheetName);

    if (!cieSheet) throw new Error("CIE sheet not found: " + cieSheetName);

    const lastRow = cieSheet.getLastRow();
    if (lastRow < 2) return [];

    // ── Dynamic header detection ──────────────────────────────────────────
    const allData = cieSheet.getRange(1, 1, lastRow, cieSheet.getLastColumn()).getValues();
    const headers = allData[0];

    const colMap = {};
    for (let i = 0; i < headers.length; i++) {
      const h = String(headers[i]).trim().toUpperCase();
      if (h === "REGD. NO."    || h === "REGD NO"    || h === "REGD.NO."   || h === "REG NO")       colMap.regnNo    = i;
      if (h === "PAPER CODE"   || h === "COURSE CODE" || h === "COURSE CODE (CC)")                   colMap.paperCode = i;
      if (h === "CIE MARKS"    || h === "MARKS"        || h === "CIE MARK")                          colMap.marks     = i;
      if (h === "CIE PERCENT"  || h === "PERCENT"      || h === "CIE %"    || h === "CIE PERCENTAGE") colMap.percent   = i;
      if (h === "CIE NP"       || h === "CIE NUM. POINTS" || h === "CIE NUM POINTS" || h === "NP")   colMap.numPoints = i;
      if (h === "PROGRAMME"    || h === "PROGRAM"      || h === "PROGRAMME CODE")                    colMap.programme = i;
      if (h === "SEMESTER"     || h === "SEM")                                                        colMap.semester  = i;
      if (h === "CAMPUS")                                                                             colMap.campus    = i;
    }

    // Validate required columns
    const required = ["regnNo", "paperCode", "marks", "programme", "semester"];
    const missing  = required.filter(k => colMap[k] === undefined);
    if (missing.length > 0) {
      Logger.log("CIE headers found: " + headers.join(", "));
      throw new Error("CIE sheet missing required columns: " + missing.join(", "));
    }

    Logger.log("✓ CIE column map: " + JSON.stringify(colMap));

    const records = [];

    for (let i = 1; i < allData.length; i++) {
      const row       = allData[i];
      const raw       = row[colMap.regnNo];
      const regnNo    = String(/^\d+(\.\d+)?([eE][+\-]?\d+)?$/.test(String(raw)) ? Math.round(Number(raw)) : raw).trim();
      const paperCode = String(row[colMap.paperCode] || "").trim().replace(/_IHA$/i, "");

      if (!regnNo || !paperCode) continue;

      const marks     = row[colMap.marks];
      const percent   = colMap.percent   !== undefined ? row[colMap.percent]   : "";
      const numPoints = colMap.numPoints !== undefined ? row[colMap.numPoints] : "";
      const programme = String(row[colMap.programme] || "").trim();
      const semester  = String(row[colMap.semester]  || "").trim();
      const campus    = colMap.campus    !== undefined ? String(row[colMap.campus] || "").trim() : "";

      if (!_matchesProgramFilter(programme, programFilter)) continue;

      records.push({
        regnNo, campus, paperCode, marks,
        percent, numPoints, programme, semester,
        assessmentType: "CIE"
      });
    }

    Logger.log("✓ Fetched " + records.length + " CIE records");
    return records;

  } catch (error) {
    Logger.log("ERROR fetching CIE marks: " + error.message);
    throw error;
  }
}

// SECTION 4: MERGE CIE AND ESE DATA
// ════════════════════════════════════════════════════════════════════════════

/**
 * Merge CIE and ESE records on regnNo + paperCode
 * @param {Array} cieRecords
 * @param {Array} eseRecords
 * @return {Array} Merged records array
 */
function mergeCIEAndESEData(cieRecords, eseRecords) {
  try {
    // Build ESE hash for O(1) lookup
    const eseHash = {};
    for (const record of eseRecords) {
      const key = record.regnNo + "|" + record.paperCode;
      eseHash[key] = record;
    }
    
    const mergedRecords = [];
    let bothCount = 0, cieOnlyCount = 0, eseOnlyCount = 0;
    
    // Process CIE records
    for (const cie of cieRecords) {
      const key = cie.regnNo + "|" + cie.paperCode;
      const ese = eseHash[key];
      
      if (ese) {
        mergedRecords.push({
          key,
          regnNo:         cie.regnNo,
          paperCode:      cie.paperCode,
          programme:      cie.programme,
          semester:       cie.semester,
          campus:         cie.campus,
          cieMarks:       cie.marks,
          eseMarks:       ese.marks,
          assessmentType: "BOTH"
        });
        bothCount++;
        delete eseHash[key];
      } else {
        mergedRecords.push({
          key,
          regnNo:         cie.regnNo,
          paperCode:      cie.paperCode,
          programme:      cie.programme,
          semester:       cie.semester,
          campus:         cie.campus,
          cieMarks:       cie.marks,
          eseMarks:       "",
          assessmentType: "CIE_ONLY"
        });
        cieOnlyCount++;
      }
    }
    
    // Add remaining ESE-only records
    for (const key in eseHash) {
      const ese = eseHash[key];
      mergedRecords.push({
        key,
        regnNo:         ese.regnNo,
        paperCode:      ese.paperCode,
        programme:      ese.programme,
        semester:       ese.semester,
        campus:         ese.campus,
        cieMarks:       "",
        eseMarks:       ese.marks,
        assessmentType: "ESE_ONLY"
      });
      eseOnlyCount++;
    }
    
    Logger.log("✓ Merge: BOTH=" + bothCount + ", CIE_ONLY=" + cieOnlyCount + ", ESE_ONLY=" + eseOnlyCount + ", Total=" + mergedRecords.length);
    
    return mergedRecords;
    
  } catch (error) {
    logEvent("ERROR", "2", "Merge failed: " + error.message, "✗");
    throw error;
  }
}

/**
 * Internal — check if a programme matches the filter
 */
function _matchesProgramFilter(programme, programFilter) {
  if (!programFilter || programFilter === "ALL") return true;
  return programFilter.split("/").some(p => p.trim() === programme);
}

/**
 * Get last row with actual data in column A
 * Ignores formula-only rows in other columns
 * @param {Sheet} sheet
 * @return {number} Last row number with data in column A
 */
function _getLastDataRow(sheet) {
  const colA = sheet.getRange("A:A").getValues();
  for (let i = colA.length - 1; i >= 1; i--) {
    if (String(colA[i][0]).trim() !== "") {
      return i + 1; // 1-indexed
    }
  }
  return 1; // Only header row exists
}

/**
 * Populate Consolidated Result — INCREMENTAL MODE
 * Only appends new records to columns A-F
 * NEVER touches columns G onwards — formulas always preserved
 */
function populateConsolidatedResult(existingRecordsHash, mergedData) {
  try {
    logEvent("DATA_POPULATOR", "3", "Starting Consolidated Result population (INCREMENTAL)...", "ℹ");
    
    const startTime = Date.now();
    const ss        = SpreadsheetApp.getActiveSpreadsheet();
    const sheet     = ss.getSheetByName(getProcessorSheetName());
    
    if (!sheet) throw new Error("Consolidated Result sheet not found");
    if (!mergedData || Object.keys(mergedData).length === 0) throw new Error("No merged data available");
    
    // ── Filter new records ────────────────────────────────────────────────
    const newRecords     = [];
    const newRecordsKeys = [];
    
    for (const key in mergedData) {
      if (!existingRecordsHash[key]) {
        newRecords.push(mergedData[key]);
        newRecordsKeys.push(key);
      }
    }
    
    if (newRecords.length === 0) {
      logEvent("DATA_POPULATOR", "3", "No new records to add", "ⓘ");
      return { success: true, newRecordsAdded: 0, newRecordsKeys: [], duration: 0 };
    }
    
    logEvent("DATA_POPULATOR", "3", "Found " + newRecords.length + " new records", "ℹ");
    
    // ── Sort ──────────────────────────────────────────────────────────────
    const sortStart = Date.now();
    newRecords.sort((a, b) => {
      if (a.regnNo !== b.regnNo) return a.regnNo.localeCompare(b.regnNo);
      return a.paperCode.localeCompare(b.paperCode);
    });
    const sortDuration = Math.round((Date.now() - sortStart) / 1000);
    
    // ── Build batch values ────────────────────────────────────────────────
    const buildStart     = Date.now();
    const batchValues    = [];
    let suffixCheckCount = 0;
    
    for (const record of newRecords) {
      const suffixStatus = _checkPaperCodeSuffix(record.paperCode);
      if (suffixStatus === "CHECK") suffixCheckCount++;
      
      batchValues.push([
        record.regnNo,
        record.paperCode,
        _safeMark(record.cieMarks),
        _safeMark(record.eseMarks),
        "",           // E: ESE after Graced Marks — formula will populate
        suffixStatus  // F: Check Programme-Semester
      ]);
    }
    const buildDuration = Math.round((Date.now() - buildStart) / 1000);
    
    // ── Find last data row using column A (ignores formula-only rows) ─────
    const lastDataRow    = _getLastDataRow(sheet);
    const startRowForNew = lastDataRow + 1;
    
    logEvent("DATA_POPULATOR", "3", "Last data row: " + lastDataRow + ", writing from row: " + startRowForNew, "ℹ");
    
    // ── Write in batches of 500 to columns A-F ONLY ───────────────────────
    const writeStart = Date.now();
    const BATCH_SIZE = 500;
    let rowsWritten  = 0;
    
    for (let i = 0; i < batchValues.length; i += BATCH_SIZE) {
      const batch      = batchValues.slice(i, Math.min(i + BATCH_SIZE, batchValues.length));
      const batchStart = startRowForNew + i;
      
      // Write ONLY columns A-F (6 columns) — never touch G+
      sheet.getRange(batchStart, 1, batch.length, 6).setValues(batch);
      rowsWritten += batch.length;
      
      const pct = Math.round((rowsWritten / batchValues.length) * 100);
      logEvent("DATA_POPULATOR", "3", "Written " + pct + "% (" + rowsWritten + "/" + batchValues.length + " rows)", "ℹ");
    }
    
    const writeDuration = Math.round((Date.now() - writeStart) / 1000);
    const duration      = Math.round((Date.now() - startTime) / 1000);
    
    logEvent("DATA_POPULATOR", "3", "Incremental population completed in " + duration + "s", "✓");
    logEvent("DATA_POPULATOR", "3", "Sort=" + sortDuration + "s Build=" + buildDuration + "s Write=" + writeDuration + "s", "ℹ");
    
    return {
      success:          true,
      newRecordsAdded:  rowsWritten,
      newRecordsKeys:   newRecordsKeys,
      suffixCheckCount: suffixCheckCount,
      startRow:         startRowForNew,
      endRow:           startRowForNew + rowsWritten - 1,
      duration:         duration
    };
    
  } catch (error) {
    logEvent("ERROR", "3", "Incremental population failed: " + error.message, "✗");
    throw error;
  }
}

/**
 * Populate Consolidated Result — FULL MODE
 * Clears ONLY columns A-F and rewrites
 * NEVER touches columns G onwards — formulas always preserved
 */
function populateConsolidatedResultFull(mergedData) {
  try {
    logEvent("DATA_POPULATOR", "3", "Starting FULL Consolidated Result population...", "ℹ");
    
    const startTime = Date.now();
    const ss        = SpreadsheetApp.getActiveSpreadsheet();
    const sheet     = ss.getSheetByName(getProcessorSheetName());
    
    if (!sheet) throw new Error("Consolidated Result sheet not found");
    if (!mergedData || Object.keys(mergedData).length === 0) throw new Error("No merged data available");
    
    // ── Clear ONLY columns A-F using column A to find real last row ───────
    const lastDataRow = _getLastDataRow(sheet);
    
    if (lastDataRow > 1) {
      sheet.getRange(2, 1, lastDataRow - 1, 6).clearContent();
      logEvent("DATA_POPULATOR", "3", "Cleared columns A-F rows 2-" + lastDataRow + " (formulas in G+ untouched)", "✓");
    }
    
    // ── Sort records ──────────────────────────────────────────────────────
    const sortStart  = Date.now();
    const records    = Object.values(mergedData);
    const recordKeys = Object.keys(mergedData);
    
    records.sort((a, b) => {
      if (a.regnNo !== b.regnNo) return a.regnNo.localeCompare(b.regnNo);
      return a.paperCode.localeCompare(b.paperCode);
    });
    const sortDuration = Math.round((Date.now() - sortStart) / 1000);
    
    // ── Build batch values ────────────────────────────────────────────────
    const buildStart     = Date.now();
    const batchValues    = [];
    let suffixCheckCount = 0;
    
    for (const record of records) {
      const suffixStatus = _checkPaperCodeSuffix(record.paperCode);
      if (suffixStatus === "CHECK") suffixCheckCount++;
      
      batchValues.push([
        record.regnNo,
        record.paperCode,
        _safeMark(record.cieMarks),
        _safeMark(record.eseMarks),
        "",
        suffixStatus
      ]);
    }
    const buildDuration = Math.round((Date.now() - buildStart) / 1000);
    
    // ── Write in batches to columns A-F ONLY — never touch G+ ────────────
    const writeStart = Date.now();
    const BATCH_SIZE = 500;
    let rowsWritten  = 0;
    
    for (let i = 0; i < batchValues.length; i += BATCH_SIZE) {
      const batch      = batchValues.slice(i, Math.min(i + BATCH_SIZE, batchValues.length));
      const batchStart = 2 + i;
      
      sheet.getRange(batchStart, 1, batch.length, 6).setValues(batch);
      rowsWritten += batch.length;
      
      const pct = Math.round((rowsWritten / batchValues.length) * 100);
      logEvent("DATA_POPULATOR", "3", "Written " + pct + "% (" + rowsWritten + "/" + batchValues.length + " rows)", "ℹ");
    }
    
    const writeDuration = Math.round((Date.now() - writeStart) / 1000);
    const duration      = Math.round((Date.now() - startTime) / 1000);
    
    // NOTE: No orphan row cleanup — formulas in G+ must never be deleted
    logEvent("DATA_POPULATOR", "3", "FULL population completed in " + duration + "s", "✓");
    logEvent("DATA_POPULATOR", "3", "Sort=" + sortDuration + "s Build=" + buildDuration + "s Write=" + writeDuration + "s", "ℹ");
    
    return {
      success:          true,
      recordsAdded:     rowsWritten,
      recordKeys:       recordKeys,
      suffixCheckCount: suffixCheckCount,
      startRow:         2,
      endRow:           rowsWritten + 1,
      duration:         duration
    };
    
  } catch (error) {
    logEvent("ERROR", "3", "FULL population failed: " + error.message, "✗");
    throw error;
  }
}

/**
 * Build hash map of existing records — uses column A to find real last row
 * @return {Object} { "regnNo|paperCode": true }
 */
function buildExistingRecordsHashMap() {
  try {
    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(getProcessorSheetName());
    
    if (!sheet) {
      Logger.log("⚠ Consolidated Result not found — treating as first sync");
      return {};
    }
    
    const lastDataRow = _getLastDataRow(sheet);
    if (lastDataRow <= 1) return {};
    
    const data    = sheet.getRange(2, 1, lastDataRow - 1, 2).getValues();
    const hashMap = {};
    
    for (const row of data) {
      const regnNo    = String(row[0]).trim();
      const paperCode = String(row[1]).trim();
      if (regnNo && paperCode) hashMap[regnNo + "|" + paperCode] = true;
    }

    return hashMap;
    
  } catch (error) {
    Logger.log("Error building existing records hash: " + error.message);
    return {};
  }
}

function _validateSMFData(smfData) {
  const regnNoCol = 11;
  const nameCol   = 12;
  const campusCol = 6;
  
  const report = {
    totalRecords:  smfData.length - 1,
    validRecords:  0,
    duplicates:    [],
    missingRegnNo: [],
    missingName:   [],
    missingCampus: [],
    allIssues:     {}
  };
  
  const regnNoMap = {};
  
  for (let i = 1; i < smfData.length; i++) {
    const row    = smfData[i];
    const regnNo = String(row[regnNoCol]).trim();
    const name   = String(row[nameCol]).trim();
    const campus = String(row[campusCol]).trim();
    
    let rowHasIssue = false;
    const rowIssues = [];
    
    if (!regnNo || regnNo === "undefined") {
      report.missingRegnNo.push({ rowIndex: i + 1, name, campus });
      rowHasIssue = true;
      rowIssues.push("MISSING_REGD_NO");
    } else {
      if (regnNoMap[regnNo]) {
        const existing = report.duplicates.find(d => d.regnNo === regnNo);
        if (!existing) {
          report.duplicates.push({ regnNo, rows: [regnNoMap[regnNo]], count: 1 });
        } else {
          existing.rows.push(i + 1);
          existing.count++;
        }
        rowHasIssue = true;
        rowIssues.push("DUPLICATE");
      } else {
        regnNoMap[regnNo] = i + 1;
      }
    }
    
    if (!name || name === "undefined") {
      report.missingName.push({ rowIndex: i + 1, regnNo });
      rowHasIssue = true;
      rowIssues.push("MISSING_NAME");
    }
    
    if (!campus || campus === "undefined") {
      report.missingCampus.push({ rowIndex: i + 1, regnNo, name });
      rowHasIssue = true;
      rowIssues.push("MISSING_CAMPUS");
    }
    
    if (rowHasIssue) {
      report.allIssues[i + 1] = { types: rowIssues, regnNo, name, campus };
    } else {
      report.validRecords++;
    }
  }
  
  return report;
}

function _markErrorsInSMFSheet(smfData, validationResult) {
  const smfFileId    = getSMFFileId();
  const smfSheetName = getSMFSheetName();
  const smfFile      = getFileById(smfFileId);
  const smfSs        = SpreadsheetApp.open(smfFile);
  const smfSheet     = smfSs.getSheetByName(smfSheetName);

  const headers  = smfSheet.getRange(1, 1, 1, smfSheet.getLastColumn()).getValues()[0];
  let qualityCol = headers.indexOf("Data_Quality_Check") + 1;

  if (qualityCol === 0) {
    qualityCol = smfSheet.getLastColumn() + 1;
    smfSheet.getRange(1, qualityCol).setValue("Data_Quality_Check");
  }

  const lastRow    = smfSheet.getLastRow();
  const dataRowCnt = lastRow - 1;
  if (dataRowCnt <= 0) return;

  // ── Build full column arrays in memory — ONE write at the end ────────────
  const values     = [];   // [["OK"], ["OK"], ...]
  const backgrounds = [];  // [["#90EE90"], ...]
  const fontColors  = [];  // [["#000000"], ...]

  for (let i = 0; i < dataRowCnt; i++) {
    values.push(["OK"]);
    backgrounds.push(["#90EE90"]);
    fontColors.push(["#000000"]);
  }

  // Overlay duplicates
  for (const dup of validationResult.duplicates) {
    for (const rowNum of dup.rows) {
      const idx = rowNum - 2;   // row 2 → index 0
      if (idx < 0 || idx >= dataRowCnt) continue;
      values[idx]      = ["DUPLICATE"];
      backgrounds[idx] = ["#FF0000"];
      fontColors[idx]  = ["#FFFFFF"];
    }
  }

  // Overlay missing regd nos
  for (const missing of validationResult.missingRegnNo) {
    const idx = missing.rowIndex - 2;
    if (idx < 0 || idx >= dataRowCnt) continue;
    values[idx]      = ["MISSING_REGD_NO"];
    backgrounds[idx] = ["#FFFF00"];
    fontColors[idx]  = ["#000000"];
  }

  // ── Three batch calls — total API cost is O(1) regardless of row count ───
  const range = smfSheet.getRange(2, qualityCol, dataRowCnt, 1);
  range.setValues(values);
  range.setBackgrounds(backgrounds);
  range.setFontColors(fontColors);

  Logger.log("✓ Marked " + Object.keys(validationResult.allIssues).length +
    " rows with errors (batched write — " + dataRowCnt + " rows, 3 API calls)");
}

function _buildSMFLookupMap(smfData, validationResult) {
  const regnNoCol = 11;
  const nameCol   = 12;
  const campusCol = 6;
  const lookupMap = {};
  
  for (let i = 1; i < smfData.length; i++) {
    const rowNum = i + 1;
    if (validationResult.allIssues[rowNum]) continue;
    
    const row    = smfData[i];
    const regnNo = String(row[regnNoCol]).trim();
    
    lookupMap[regnNo] = {
      name:     String(row[nameCol]).trim().replace(/\s+/g, " "),
      campus:   String(row[campusCol]).trim(),
      rowIndex: rowNum
    };
  }
  
  Logger.log("✓ Built SMF lookup map with " + Object.keys(lookupMap).length + " valid records");
  return lookupMap;
}

function _createSMFValidationReport(validationResult) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let reportSheet = ss.getSheetByName("_SMFValidationReport");
  
  if (!reportSheet) {
    reportSheet = ss.insertSheet("_SMFValidationReport");
  } else {
    reportSheet.clear();
  }
  
  reportSheet.getRange("A1").setValue("SMF DATA VALIDATION REPORT")
    .setFontSize(14).setBackground("#4285F4").setFontColor("#FFFFFF");
  
  reportSheet.getRange("A3").setValue("SUMMARY")
    .setFontSize(12).setBackground("#E8F0FE");
  
  const summary = [
    ["Total Records",   validationResult.totalRecords],
    ["Valid Records",   validationResult.validRecords],
    ["Duplicates",      validationResult.duplicates.length],
    ["Missing Regd No", validationResult.missingRegnNo.length],
    ["Missing Names",   validationResult.missingName.length],
    ["Missing Campus",  validationResult.missingCampus.length],
    ["Data Quality",    Math.round((validationResult.validRecords / validationResult.totalRecords) * 100) + "%"]
  ];
  
  reportSheet.getRange(4, 1, summary.length, 2).setValues(summary);
  reportSheet.setColumnWidth(1, 150);
  reportSheet.setColumnWidth(2, 250);
  
  Logger.log("✓ Created SMF validation report");
}

// CMF WEIGHTAGE READER
// Single CMF open — builds one map used by both Rule 3 Part A and Part B
// ════════════════════════════════════════════════════════════════════════════

function readCMFWeightages() {
  try {
    logEvent("COMPLETENESS", "V", "Reading CMF weightages and max marks...", "ℹ");

    const cmfFileId    = getCMFFileId();
    const cmfSheetName = getCMFSheetName();
    const cmfFile      = SpreadsheetApp.openById(cmfFileId);
    const cmfSheet     = cmfFile.getSheetByName(cmfSheetName);

    if (!cmfSheet) throw new Error("CMF sheet '" + cmfSheetName + "' not found");

    const lastRow = cmfSheet.getLastRow();
    if (lastRow < 2) throw new Error("CMF sheet is empty");

    const allData = cmfSheet.getRange(1, 1, lastRow, cmfSheet.getLastColumn()).getValues();
    const headers = allData[0];

    Logger.log("CMF Headers: " + headers.join(", "));

    const colMap = {};
    for (let i = 0; i < headers.length; i++) {
      const h = String(headers[i]).trim().toUpperCase();

      if (h === "BATCH")                                                   colMap.batch        = i;
      if (h === "CIE WTG")                                                 colMap.cieWtg       = i;
      if (h === "ESE WTG")                                                 colMap.eseWtg       = i;
      if (h === "CIE MAX" || h === "CIE MARKS")                            colMap.cieMax       = i;
      if (h === "ESE MAX" || h === "ESE MARKS")                            colMap.eseMax       = i;
      if (h === "COURSE CODE" || h === "COURSE CODE (CC)")                 colMap.courseCode   = i;
      if (h === "PROGRAMME CODE")                                          colMap.programCode  = i;
      if (h === "SEMESTER")                                                colMap.semester     = i;
      if (h === "TEACHING DEPARTMENT")                                     colMap.teachingDept = i;
      if (h === "CREDITS" || h === "CREDIT" || h === "COURSE CREDITS")    colMap.credits      = i;
      if (h === "SWAYAM-ASSIGN" || h === "SWAYAM ASSIGN")                 colMap.swayamAssign = i;
      if (h === "SWAYAM-LINK"   || h === "SWAYAM LINK")                   colMap.swayamLink   = i;
    }

    Logger.log("CMF column map: " + JSON.stringify(colMap));

    const required = ["cieWtg", "eseWtg", "cieMax", "eseMax", "courseCode", "programCode"];
    const missing  = required.filter(k => colMap[k] === undefined);
    if (missing.length > 0) {
      Logger.log("Available headers: " + headers.map((h, i) => i + ":" + h).join(", "));
      throw new Error("CMF missing required columns: " + missing.join(", "));
    }

    if (colMap.batch === undefined) {
      Logger.log("⚠ CMF has no Batch column — batch validation will be skipped");
    }

    const weightageMap = {};
    let   skipped      = 0;
    let   swayamCount  = 0;

    const _parse = v => {
      if (v === "" || v === null || v === undefined) return null;
      if (String(v).trim().toUpperCase() === "NC") return null;  // NC = not counted, no marks expected
      return Number(v);
    };

    for (let i = 1; i < allData.length; i++) {
      const row         = allData[i];
      const courseCode  = String(row[colMap.courseCode]  || "").trim();
      const programCode = String(row[colMap.programCode] || "").trim();
      const semester    = colMap.semester     !== undefined ? String(row[colMap.semester]     || "").trim() : "";
      const teachDept   = colMap.teachingDept !== undefined ? String(row[colMap.teachingDept] || "").trim().toUpperCase() : "";
      const batch       = colMap.batch        !== undefined ? String(row[colMap.batch]        || "").trim() : "";

      if (!courseCode) { skipped++; continue; }

      // Skip _IHA variants — IHA marks arrive under base course code after stripping
      if (courseCode.toUpperCase().endsWith("_IHA")) { skipped++; continue; }

      const isExternalSWAYAM = teachDept === "SWAYAM" || teachDept.includes("SWAYAM PORTAL");
      const isInhouseSWAYAM  = teachDept.includes("SWAYAM INHOUSE");

      if (isExternalSWAYAM || (!isInhouseSWAYAM && _isNPTELCourse(courseCode))) {
        swayamCount++;
        continue;
      }

      if (courseCode.toUpperCase().endsWith("_ASSIGN")) { skipped++; continue; }

      if (!weightageMap[courseCode]) {
        weightageMap[courseCode] = {
          courseCode,
          programCode,
          semester,
          batch,
          isBEd:        _isBEdProgramme(programCode),
          cieWtg:       _parse(row[colMap.cieWtg]),
          eseWtg:       _parse(row[colMap.eseWtg]),
          cieMax:       _parse(row[colMap.cieMax]),
          eseMax:       _parse(row[colMap.eseMax]),
          teachingDept: teachDept
        };
      }
    }

    logEvent("COMPLETENESS", "V",
      "CMF weightage map: " + Object.keys(weightageMap).length + " courses" +
      " (" + skipped + " skipped, " + swayamCount + " SWAYAM excluded)", "✓");

    return weightageMap;

  } catch (error) {
    logEvent("ERROR", "V", "readCMFWeightages failed: " + error.message, "✗");
    throw error;
  }
}

function _isBEdProgramme(programCode) {
  const code = String(programCode).toUpperCase().replace(/[\s.\-]/g, "");
  return code === "BED" || code.startsWith("BED") || code.endsWith("BED");
}

/**
 * Run all three completeness rules
 * @param  {Array}  cieRecords         — from fetchCIEMarks()
 * @param  {Array}  eseRecords         — from fetchESEMarks()
 * @param  {Object} academicYearLookup — { regnNo → { programme, semester, name } }
 * @return {Object} {
 *   eligibleProgrammes : Set<"programme|semester">,
 *   blockedProgrammes  : { "programme|semester" → { reasons[], details{} } },
 *   leftStudents       : { regnNo → { detectedIn, paperCodes[] } },
 *   rule1, rule2, rule3, duration
 * }
 */



// ════════════════════════════════════════════════════════════════════════════
// PRE-PROCESS: IDENTIFY LEFT STUDENTS
// ════════════════════════════════════════════════════════════════════════════

function _identifyLeftStudents(cieRecords, eseRecords) {
  const left = {};

  const _mark = (regnNo, paperCode, source) => {
    if (!left[regnNo]) left[regnNo] = { detectedIn: new Set(), paperCodes: [] };
    left[regnNo].detectedIn.add(source);
    if (!left[regnNo].paperCodes.includes(paperCode)) {
      left[regnNo].paperCodes.push(paperCode);
    }
  };

  for (const r of cieRecords) {
    if (String(r.marks).trim().toUpperCase() === "L") _mark(r.regnNo, r.paperCode, "CIE");
  }
  for (const r of eseRecords) {
    if (String(r.marks).trim().toUpperCase() === "L") _mark(r.regnNo, r.paperCode, "ESE");
  }

  // Normalise Set → array, resolve BOTH
  for (const regnNo in left) {
    const d = left[regnNo].detectedIn;
    left[regnNo].detectedIn = (d.has("CIE") && d.has("ESE")) ? ["BOTH"] : [...d];
  }

  return left;
}


// ════════════════════════════════════════════════════════════════════════════
// RULE 1: STUDENT COUNT
// ════════════════════════════════════════════════════════════════════════════

function _validateStudentCount(cieRecords, eseRecords, academicYearLookup, excludedRegnNos) {

  // ═══════════════════════════════════════════════════════════════════
  // FIX — Absent student handling
  // ---------------------------------------------------------------
  // If a student is marked "A" (absent) in the CIE fetch file, the ESE
  // fetch file often has NO row for them at all (they never sat the exam).
  // Without this pre-step they fall into missingInESE → missingRegnNos
  // → excludedRegnNos, and their "A" CIE records never reach the
  // consolidated result sheet.
  //
  // Resolution: treat a student with ANY "A" mark in CIE as "documented
  // in ESE" so they are never flagged as missing in the opposite set.
  // Symmetric logic handles students with "A" in ESE but absent from CIE.
  // ═══════════════════════════════════════════════════════════════════

  // Per-progKey Sets of students with at least one "A" mark
  const absentInCIEByProgKey = {};   // progKey → Set<regnNo>
  const absentInESEByProgKey = {};   // progKey → Set<regnNo>

  for (const r of cieRecords) {
    if (excludedRegnNos.has(r.regnNo)) continue;
    if (String(r.marks === null || r.marks === undefined ? "" : r.marks)
          .trim().toUpperCase() !== "A") continue;

    const s = academicYearLookup[r.regnNo];
    if (!s) continue;
    const key = s.programme + "|" + s.semester + "|" + (s.batch || "");
    if (!absentInCIEByProgKey[key]) absentInCIEByProgKey[key] = new Set();
    absentInCIEByProgKey[key].add(r.regnNo);
  }

  for (const r of eseRecords) {
    if (excludedRegnNos.has(r.regnNo)) continue;
    if (String(r.marks === null || r.marks === undefined ? "" : r.marks)
          .trim().toUpperCase() !== "A") continue;

    const s = academicYearLookup[r.regnNo];
    if (!s) continue;
    const key = s.programme + "|" + s.semester + "|" + (s.batch || "");
    if (!absentInESEByProgKey[key]) absentInESEByProgKey[key] = new Set();
    absentInESEByProgKey[key].add(r.regnNo);
  }

  // ── Original SMF sets ─────────────────────────────────────────────────
  const smfSets = {};
  for (const regnNo in academicYearLookup) {
    if (excludedRegnNos.has(regnNo)) continue;
    const s   = academicYearLookup[regnNo];
    const key = s.programme + "|" + s.semester + "|" + (s.batch || "");
    if (!smfSets[key]) smfSets[key] = new Set();
    smfSets[key].add(regnNo);
  }

  // ── CIE arrived sets ──────────────────────────────────────────────────
  const cieSets = {};
  for (const r of cieRecords) {
    if (excludedRegnNos.has(r.regnNo)) continue;
    const s = academicYearLookup[r.regnNo];
    if (!s) continue;
    const key = s.programme + "|" + s.semester + "|" + (s.batch || "");
    if (!cieSets[key]) cieSets[key] = new Set();
    cieSets[key].add(r.regnNo);
  }

  // ── Patch: students absent in ESE are already documented — count as arrived in CIE ──
  for (const key in absentInESEByProgKey) {
    if (!cieSets[key]) cieSets[key] = new Set();
    for (const regnNo of absentInESEByProgKey[key]) {
      cieSets[key].add(regnNo);
    }
  }

  // ── ESE arrived sets ──────────────────────────────────────────────────
  const eseSets = {};
  for (const r of eseRecords) {
    if (excludedRegnNos.has(r.regnNo)) continue;
    const s = academicYearLookup[r.regnNo];
    if (!s) continue;
    const key = s.programme + "|" + s.semester + "|" + (s.batch || "");
    if (!eseSets[key]) eseSets[key] = new Set();
    eseSets[key].add(r.regnNo);
  }

  // ── Patch: students absent in CIE are already documented — count as arrived in ESE ──
  for (const key in absentInCIEByProgKey) {
    if (!eseSets[key]) eseSets[key] = new Set();
    for (const regnNo of absentInCIEByProgKey[key]) {
      eseSets[key].add(regnNo);
    }
  }

  // ── Rest of the function is UNCHANGED from here ───────────────────────
  const missingRegnNos    = {};
  const warningProgrammes = {};
  const passedProgrammes  = new Set();
  const failedProgrammes  = {};
  const emptyProgrammes   = new Set();

  for (const progKey in smfSets) {
    const smfSet = smfSets[progKey];
    const cieSet = cieSets[progKey] || new Set();
    const eseSet = eseSets[progKey] || new Set();

    const parts      = progKey.split("|");
    const programme  = parts[0];
    const semester   = parts[1];
    const batch      = parts[2] || "";

    const missingInCIE    = [...smfSet].filter(r => !cieSet.has(r));
    const missingInESE    = [...smfSet].filter(r => !eseSet.has(r));
    const unexpectedInCIE = [...cieSet].filter(r => !smfSet.has(r));
    const unexpectedInESE = [...eseSet].filter(r => !smfSet.has(r));

    for (const regnNo of smfSet) {
      const missingCIE = !cieSet.has(regnNo);
      const missingESE = !eseSet.has(regnNo);

      if (missingCIE || missingESE) {
        if (!missingRegnNos[regnNo]) {
          missingRegnNos[regnNo] = { regnNo, programme, semester, batch, missingIn: [] };
        }
        if (missingCIE) missingRegnNos[regnNo].missingIn.push("CIE");
        if (missingESE) missingRegnNos[regnNo].missingIn.push("ESE");
      }
    }

    const hasRealIssue = missingInCIE.length > 0 || unexpectedInCIE.length > 0;
    const hasEseIssue  = missingInESE.length > 0 || unexpectedInESE.length > 0;

    if (hasRealIssue || hasEseIssue) {
      failedProgrammes[progKey] = {
        progKey,
        smfCount:         smfSet.size,
        adjustedExpected: smfSet.size,
        cieCount:         cieSet.size,
        eseCount:         eseSet.size,
        missingInCIE,
        missingInESE,
        unexpectedInCIE,
        unexpectedInESE
      };
    } else if (hasEseIssue) {
      warningProgrammes[progKey] = {
        progKey,
        smfCount:         smfSet.size,
        adjustedExpected: smfSet.size,
        cieCount:         cieSet.size,
        eseCount:         eseSet.size,
        missingInCIE:     [],
        missingInESE,
        unexpectedInCIE:  [],
        unexpectedInESE
      };
    }

    passedProgrammes.add(progKey);
  }

  for (const progKey in smfSets) {
    const smfSet         = smfSets[progKey];
    const remainingCount = [...smfSet].filter(r => !missingRegnNos[r]).length;

    if (remainingCount === 0 && smfSet.size > 0) {
      emptyProgrammes.add(progKey);
      Logger.log("⚠ Programme with ALL students excluded: " + progKey +
        " (SMF: " + smfSet.size + ", remaining: 0)");
    }
  }

  Logger.log("✓ Rule 1A → missing students: "   + Object.keys(missingRegnNos).length +
    " | failed programmes: "                     + Object.keys(failedProgrammes).length +
    " | warning programmes: "                    + Object.keys(warningProgrammes).length +
    " | empty programmes: "                      + emptyProgrammes.size);

  return { passedProgrammes, failedProgrammes, warningProgrammes, missingRegnNos, emptyProgrammes, smfSets, cieSets, eseSets };
}



// ════════════════════════════════════════════════════════════════════════════
// RULE 2: MARKS QUALITY
// ════════════════════════════════════════════════════════════════════════════

function _validateMarksQuality(cieRecords, eseRecords, leftRegnNos, cmfWeightageMap, excludedPaperKeys) {
  const hardBlocks = {};
  const warnings   = {};

  // ── CHANGED: resolve paper exclusion set defensively ──────────────────
  const paperExclusions = excludedPaperKeys instanceof Set ? excludedPaperKeys : new Set();

  const _check = (regnNo, paperCode, mark, markType) => {
    if (leftRegnNos.has(regnNo)) return;

    // ── CHANGED: skip paper-level overrides ───────────────────────────────
     if (paperExclusions.has(paperCode)) return;

    const markStr = String(mark === null || mark === undefined ? "" : mark).trim();
    const upper   = markStr.toUpperCase();
    const key     = regnNo + "|" + paperCode + "|" + markType;

    if (upper === "A") return;

    if (markStr === "") {
      hardBlocks[key] = { regnNo, paperCode, markType, mark: "", flag: "BLANK_" + markType };
      return;
    }

    const numMark = Number(markStr);

    if (isNaN(numMark)) {
      hardBlocks[key] = { regnNo, paperCode, markType, mark: markStr, flag: "UNRECOGNISED_" + markType };
      return;
    }

    if (numMark === 0) {
      warnings[key] = { regnNo, paperCode, markType, mark: 0, flag: "ZERO_" + markType };
      return;
    }

    const cmf        = cmfWeightageMap ? cmfWeightageMap[paperCode] : null;
    const maxAllowed = cmf ? (markType === "CIE" ? cmf.cieMax : cmf.eseMax) : null;

    if (maxAllowed !== null && !isNaN(maxAllowed) && maxAllowed > 0) {
      const pct = (numMark / maxAllowed) * 100;
      if (pct < 20) {
        warnings[key] = {
          regnNo, paperCode, markType,
          mark: numMark, maxAllowed,
          pct:  Math.round(pct),
          flag: "LOW_" + markType
        };
      }
    }
  };

  for (const r of cieRecords) _check(r.regnNo, r.paperCode, r.marks, "CIE");
  for (const r of eseRecords) _check(r.regnNo, r.paperCode, r.marks, "ESE");

  return { flaggedRows: hardBlocks, warnings };
}

function _validateUniformMarks(cieRecords, eseRecords, excludedRegnNos, academicYearLookup) {

  // groups[paperCode|CIE] = { paperCode, markType, programme, semester, marks[], regnNos[] }
  const groups = {};

  const _collect = (records, markType) => {
    for (const r of records) {
      if (excludedRegnNos.has(r.regnNo)) continue;

      const markStr = String(r.marks === null || r.marks === undefined ? "" : r.marks).trim().toUpperCase();

      // Skip absent, left, blank — only numeric marks count toward uniformity
      if (markStr === "A" || markStr === "L" || markStr === "") continue;

      const numMark = Number(markStr);
      if (isNaN(numMark)) continue;  // Rule 2 already flags these

      const student = academicYearLookup[r.regnNo];
      if (!student) continue;

      const key = r.paperCode + "|" + markType;
      if (!groups[key]) {
        groups[key] = {
          paperCode:  r.paperCode,
          markType,
          programme:  student.programme,
          semester:   student.semester,
          marks:      [],
          regnNos:    []
        };
      }
      groups[key].marks.push(numMark);
      groups[key].regnNos.push(r.regnNo);
    }
  };

  _collect(cieRecords, "CIE");
  _collect(eseRecords, "ESE");

  const warnings = {};

  for (const key in groups) {
    const g = groups[key];

    // Need at least 2 students to call it uniform
    if (g.marks.length < 2) continue;

    const allSame = g.marks.every(m => m === g.marks[0]);
    if (!allSame) continue;

    const uniformMark = g.marks[0];

    // Edge case: all zeros are already caught by Rule 2 ZERO warnings.
    // Still flag here too — uniform zero across a full class is especially suspicious.
    warnings[key] = {
      paperCode:     g.paperCode,
      markType:      g.markType,
      programme:     g.programme,
      semester:      g.semester,
      uniformMark,
      studentCount:  g.marks.length,
      flag:          "UNIFORM_" + g.markType + "_MARKS",
      note:          uniformMark === 0
                       ? "All students scored zero — likely a missed entry"
                       : "All students scored " + uniformMark + " — verify with examiner"
    };
  }

  logEvent("COMPLETENESS", "V",
    "Rule 4 → uniform marks warnings: " + Object.keys(warnings).length, "ℹ");

  return { warnings };
}

function _classifyCourse(courseCategory) {
  const cat = String(courseCategory || "").trim().toUpperCase();
  if (cat === "DSC")                                                    return "CORE";
  if (cat === "DSE")                                                    return "ELECTIVE";
  if (["IDM", "SEC", "AEC", "MDC", "VBC", "VOC", "SEC"].includes(cat)) return cat;
  return "CORE"; // default — unknown categories treated as core (strict)
}

function _isSpecialCategory(category) {
  return ["ELECTIVE", "IDM", "SEC", "AEC", "MDC", "VBC", "VOC", "SSC"].includes(category);
}

function _readSMFCategoryEnrollments(excludedRegnNos) {
  try {
    const smfFileId    = getSMFFileId();
    const smfSheetName = getSMFSheetName();
    const smfFile      = getFileById(smfFileId);
    const smfSs        = SpreadsheetApp.open(smfFile);
    const smfSheet     = smfSs.getSheetByName(smfSheetName);

    if (!smfSheet) throw new Error("SMF sheet not found: " + smfSheetName);

    const allData = smfSheet.getDataRange().getValues();
    if (allData.length < 2) {
      Logger.log("ℹ SMF is empty — no category enrollment data");
      return { courseEnrollment: {}, categoryColumns: {}, found: false };
    }

    const headers = allData[0];

    // ── Identify regnNo column ────────────────────────────────────────────
    let regnNoCol = -1;
    for (let i = 0; i < headers.length; i++) {
      const h = String(headers[i] || "").trim().toUpperCase();
      if (h === "REGD. NO." || h === "REGD NO" || h === "REGD.NO." || h === "REG NO") {
        regnNoCol = i;
        break;
      }
    }
    if (regnNoCol === -1) throw new Error("SMF: Regd No column not found in headers");

    // ── Find all columns whose header contains a special-category keyword ──
    // e.g. header "IDM Course" matches keyword "IDM"
    // Allow one column per keyword — last match wins if duplicates exist
    const KEYWORDS = ["IDM", "SEC", "DSE", "AEC", "MDC", "VBC", "VOC", "SEC", "ELECTIVE"];
    const categoryColumns = {};

    for (let i = 0; i < headers.length; i++) {
      const h = String(headers[i] || "").trim().toUpperCase();
      for (const kw of KEYWORDS) {
        if (h.includes(kw)) {
          // Normalise: ELECTIVE and DSE both map to "ELECTIVE"
          const normKey = (kw === "DSE") ? "ELECTIVE" : kw;
          if (!categoryColumns[normKey]) {
            categoryColumns[normKey] = i;
            Logger.log("✓ SMF category column: [" + headers[i] + "] col=" + i + " → key=" + normKey);
          }
        }
      }
    }

    if (Object.keys(categoryColumns).length === 0) {
      Logger.log("ℹ SMF has no category columns (IDM / SEC / DSE / AEC / MDC / VBC / VOC) — " +
        "per-course enrollment unavailable; special-category counts will use total class size");
      return { courseEnrollment: {}, categoryColumns: {}, found: false };
    }

    // ── Build courseEnrollment: courseCode → Set<regnNo> ──────────────────
    const courseEnrollment = {};

    for (let i = 1; i < allData.length; i++) {
      const row    = allData[i];
      const regnNo = String(row[regnNoCol] || "").trim();

      if (!regnNo || excludedRegnNos.has(regnNo)) continue;

      for (const normKey in categoryColumns) {
        const colIdx     = categoryColumns[normKey];
        const courseCode = String(row[colIdx] || "").trim();

        // Blank = student not enrolled in any course for this category
        if (!courseCode) continue;

        if (!courseEnrollment[courseCode]) courseEnrollment[courseCode] = new Set();
        courseEnrollment[courseCode].add(regnNo);
      }
    }

    Logger.log("✓ SMF category enrollment: " + Object.keys(courseEnrollment).length +
      " courses with per-student enrollment" +
      " (categories found: " + Object.keys(categoryColumns).join(", ") + ")");

    return { courseEnrollment, categoryColumns, found: true };

  } catch (error) {
    logEvent("WARN", "V",
      "_readSMFCategoryEnrollments failed — " + error.message +
      " — falling back to total class count for special categories", "⚠");
    return { courseEnrollment: {}, categoryColumns: {}, found: false };
  }
}



// ════════════════════════════════════════════════════════════════════════════
// RULE 3: DISTRIBUTION
// ════════════════════════════════════════════════════════════════════════════

function _validateDistribution(cieRecords, eseRecords, cmfWeightageMap, excludedRegnNos, excludedPaperKeys) {

  // ── CHANGED: resolve paper exclusion set defensively ──────────────────
  const paperExclusions = excludedPaperKeys instanceof Set ? excludedPaperKeys : new Set();

  // ── Part A: Weightage ratio check — unchanged; operates only on CMF ────
  const partAHard     = [];
  const partAWarnings = [];

  for (const courseCode in cmfWeightageMap) {
    const c = cmfWeightageMap[courseCode];

    if (c.cieWtg === null || isNaN(c.cieWtg) ||
        c.eseWtg === null || isNaN(c.eseWtg)) {
      partAHard.push({
        courseCode: c.courseCode, programCode: c.programCode,
        cieWtg: c.cieWtg, eseWtg: c.eseWtg, cieMax: c.cieMax, eseMax: c.eseMax,
        flags: ["MISSING_WTG"]
      });
      continue;
    }

    const totalWtg = c.cieWtg + c.eseWtg;
    if (totalWtg === 0) continue;

    if (Math.round(totalWtg) !== 100) {
      partAHard.push({
        courseCode: c.courseCode, programCode: c.programCode,
        cieWtg: c.cieWtg, eseWtg: c.eseWtg, totalWtg,
        cieMax: c.cieMax, eseMax: c.eseMax,
        flags: ["INVALID_TOTAL_WTG"]
      });
      continue;
    }

    if (c.eseWtg === 0 || c.cieWtg === 0) continue;

    const expectedCIE = c.isBEd ? 30 : 50;
    const expectedESE = c.isBEd ? 70 : 50;

    if (c.cieWtg !== expectedCIE || c.eseWtg !== expectedESE) {
      partAWarnings.push({
        courseCode:     c.courseCode,
        programCode:    c.programCode,
        isBEd:          c.isBEd,
        cieWtg:         c.cieWtg,
        eseWtg:         c.eseWtg,
        expectedCIEWtg: expectedCIE,
        expectedESEWtg: expectedESE,
        flags:          ["WRONG_WEIGHTAGE"]
      });
    }
  }

  // ── Part B: Submitted marks vs CMF Max ───────────────────────────────────
  const partBHard           = [];
  const partBWarnings       = [];
  const unknownPaperCodes   = new Set();
  const unknownPaperRecords = [];

  const _checkMax = (regnNo, paperCode, mark, markType) => {
    if (excludedRegnNos.has(regnNo))                         return;
    if (_isPhDPaperCode(paperCode))                          return;
    if (_isNPTELCourse(paperCode))                           return;

    // ── CHANGED: skip paper-level overrides ─────────────────────────────
    if (paperExclusions.has(paperCode))                      return;

    const markStr = String(mark === null || mark === undefined ? "" : mark).trim();
    const upper   = markStr.toUpperCase();
    if (upper === "A" || upper === "L" || markStr === "")    return;

    const numMark = Number(markStr);
    if (isNaN(numMark))                                      return;

    const cmf = cmfWeightageMap[paperCode];

    if (!cmf) {
      unknownPaperCodes.add(paperCode);
      unknownPaperRecords.push({ regnNo, paperCode, markType, submittedMark: numMark });
      logEvent("COMPLETENESS", "V",
        "UNKNOWN_PAPER_CODE — record filtered, student continues: " +
        regnNo + " | " + paperCode + " | " + markType, "⚠");
      return;
    }

    const maxAllowed = markType === "CIE" ? cmf.cieMax : cmf.eseMax;
    if (maxAllowed === null || isNaN(maxAllowed)) return;

    if (maxAllowed === 0 && numMark > 0) {
      partBWarnings.push({
        regnNo, paperCode, markType, submittedMark: numMark,
        maxAllowed: 0, flag: "UNEXPECTED_" + markType + "_MARK"
      });
      return;
    }

    if (numMark > maxAllowed) {
      partBWarnings.push({
        regnNo, paperCode, markType, submittedMark: numMark,
        maxAllowed, flag: "EXCEEDS_" + markType + "_MAX"
      });
    }
  };

  for (const r of cieRecords) _checkMax(r.regnNo, r.paperCode, r.marks, "CIE");
  for (const r of eseRecords) _checkMax(r.regnNo, r.paperCode, r.marks, "ESE");

  if (unknownPaperCodes.size > 0) {
    logEvent("COMPLETENESS", "V",
      "Unknown paper codes (record-level filter only, no programme block): " +
      [...unknownPaperCodes].join(", "), "⚠");
  }

  return {
    partA: { flaggedCourses: partAHard, warnings: partAWarnings },
    partB: { flaggedRows:    partBHard, warnings: partBWarnings },
    unknownPaperCodes,
    unknownPaperRecords
  };
}



function _truncateList(arr, max) {
  if (!arr || arr.length === 0) return "None";
  const shown = arr.slice(0, max).join(", ");
  return arr.length > max
    ? shown + " (+" + (arr.length - max) + " more)"
    : shown;
}


// ── Helpers used by _validateDistribution — must be top-level ────────────

function _isPhDPaperCode(paperCode) {
  const code = String(paperCode).toUpperCase().trim();
  return code.startsWith("PPHD-") || code.includes("PPHD");
}

function _isNPTELCourse(paperCode) {
  const code = String(paperCode).toLowerCase().trim();
  return /^[a-z]{2,4}\d{2}_/.test(code) ||   // starts with prefix
         /\([a-z]{2,4}\d{2}_/.test(code);     // embedded in parentheses
}


// ════════════════════════════════════════════════════════════════════════════
// SPLIT ELIGIBLE AND BLOCKED
// ════════════════════════════════════════════════════════════════════════════

/**
 * Partition mergedData into eligibleData and blockedData.
 * Left students are dropped from both buckets.
 *
 * @param  {Object} mergedData
 * @param  {Object} blockedProgrammes  — from validateCompleteness()
 * @param  {Object} leftStudents       — from validateCompleteness()
 * @param  {Object} academicYearLookup
 * @return {Object} { eligibleData, blockedData }
 */
function splitEligibleAndBlocked(
  mergedData, blockedProgrammes, leftStudents,
  academicYearLookup, excludedRegnNos,
  unknownPaperCodes, excludedPaperKeys
) {
  const eligibleData = {};
  const blockedData  = {};
  const notInSMFData = {};

  const excluded     = excludedRegnNos instanceof Set
    ? excludedRegnNos
    : new Set(Object.keys(leftStudents));

  const unknownCodes  = unknownPaperCodes  instanceof Set ? unknownPaperCodes  : new Set();

  // ── CHANGED: paper-level exclusions ───────────────────────────────────
  const paperExclusions = excludedPaperKeys instanceof Set ? excludedPaperKeys : new Set();

  let unknownFiltered = 0;
  let paperFiltered   = 0;

  for (const key in mergedData) {
    const record    = mergedData[key];
    const regnNo    = record.regnNo;
    const paperCode = record.paperCode;

    if (excluded.has(regnNo)) continue;

    if (unknownCodes.has(paperCode)) {
      unknownFiltered++;
      logEvent("COMPLETENESS", "V",
        "Filtered unknown paper from pipeline: " + regnNo + " | " + paperCode, "⚠");
      continue;
    }

    // ── CHANGED: skip paper-level overrides — record dropped from pipeline
    if (paperExclusions.has(paperCode)) {
      paperFiltered++;
      logEvent("COMPLETENESS", "V",
        "Filtered paper override from pipeline: " + paperCode + " (all students)", "⚠");
      continue;
    }

    const student = academicYearLookup[regnNo];

    if (!student) {
      if (!notInSMFData[regnNo]) notInSMFData[regnNo] = { regnNo, records: [] };
      notInSMFData[regnNo].records.push(record);
      continue;
    }

    const progKey = student.programme + "|" + student.semester + "|" + (student.batch || "");

    if (blockedProgrammes[progKey]) {
      blockedData[key] = record;
    } else {
      eligibleData[key] = record;
    }
  }

  const notInSMFCount = Object.keys(notInSMFData).length;

  if (notInSMFCount > 0) {
    logEvent("COMPLETENESS", "V",
      notInSMFCount + " students found in fetch but NOT in SMF — parked in _NotInSMFStudents", "⚠");
  }
  if (unknownFiltered > 0) {
    logEvent("COMPLETENESS", "V",
      unknownFiltered + " records filtered — paper code not in CMF (student continues)", "⚠");
  }
  // ── CHANGED: log paper-override filtered count ─────────────────────────
  if (paperFiltered > 0) {
    logEvent("COMPLETENESS", "V",
      paperFiltered + " records filtered — paper-level validation override applied", "⚠");
  }

  logEvent("COMPLETENESS", "V",
    "Split complete → eligible: " + Object.keys(eligibleData).length +
    " | blocked: "               + Object.keys(blockedData).length  +
    " | not in SMF: "            + notInSMFCount, "✓");

  return { eligibleData, blockedData, notInSMFData };
}

/**
 * Write blocked programme students to _PendingResults.
 * Existing entries are updated (Last Checked, Runs Waited).
 * New entries are appended.
 */
function writePendingResults(blockedProgrammes, blockedData, academicYearLookup) {
  try {
    if (Object.keys(blockedProgrammes).length === 0) {
      logEvent("COMPLETENESS", "V", "No blocked programmes — _PendingResults not updated", "ℹ");
      return;
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName("_PendingResults");

    if (!sheet) {
      sheet = ss.insertSheet("_PendingResults");
      const hdr = [
        "Programme", "Semester", "Batch", "Regd No", "Name",
        "Arrived Papers", "Blocking Reasons",
        "First Blocked On", "Last Checked On", "Runs Waited", "Status"
      ];
      sheet.getRange(1, 1, 1, hdr.length).setValues([hdr]);
      const hr = sheet.getRange(1, 1, 1, hdr.length);
      hr.setFontWeight("bold").setBackground("#000000").setFontColor("#FFFFFF");
      sheet.setFrozenRows(1);
    } else {
      // ── Migrate old 10-column sheet to 11-column if needed ──────────────
      const existingHeader = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
      if (existingHeader[2] !== "Batch") {
        sheet.insertColumnAfter(2);
        sheet.getRange(1, 3).setValue("Batch").setFontWeight("bold")
          .setBackground("#000000").setFontColor("#FFFFFF");
        Logger.log("✓ _PendingResults: inserted Batch column (migration)");
      }
    }

    const lastRow      = sheet.getLastRow();
    let   existingRows = [];
    const existingMap  = {};

    if (lastRow > 1) {
      existingRows = sheet.getRange(2, 1, lastRow - 1, 11).getValues();
      existingRows.forEach((r, i) => {
        const rn = String(r[3]).trim();  // col 4 = Regd No
        if (rn && r[10] !== "RESOLVED") existingMap[rn] = i;
      });
    }

    const byStudent = {};
    for (const key in blockedData) {
      const r = blockedData[key];
      if (!byStudent[r.regnNo]) byStudent[r.regnNo] = [];
      byStudent[r.regnNo].push(r.paperCode);
    }

    const now     = getTimestamp();
    const newRows = [];

    for (const regnNo in byStudent) {
      const student = academicYearLookup[regnNo];
      if (!student) continue;

      const progKey = student.programme + "|" + student.semester + "|" + (student.batch || "");
      const blocked = blockedProgrammes[progKey];
      if (!blocked) continue;

      const reasons = blocked.reasons.join(", ");

      if (existingMap[regnNo] !== undefined) {
        const idx         = existingMap[regnNo];
        existingRows[idx][8]  = now;                                        // Last Checked On
        existingRows[idx][9]  = (Number(existingRows[idx][9]) || 1) + 1;  // Runs Waited
      } else {
        newRows.push([
          student.programme,
          student.semester,
          student.batch || "",
          regnNo,
          student.name,
          byStudent[regnNo].length,
          reasons,
          now,
          now,
          1,
          "PENDING"
        ]);
      }
    }

    if (existingRows.length > 0) {
      sheet.getRange(2, 1, existingRows.length, 11).setValues(existingRows);
    }

    if (newRows.length > 0) {
      const startRow = sheet.getLastRow() + 1;
      sheet.getRange(startRow, 1, newRows.length, 11).setValues(newRows);
      sheet.getRange(startRow, 1, newRows.length, 11).setBackground("#FFCCCC");
    }

    logEvent("COMPLETENESS", "V",
      "_PendingResults → new: " + newRows.length +
      " | updated: " + Object.keys(existingMap).length, "✓");

  } catch (error) {
    logEvent("ERROR", "V", "writePendingResults failed: " + error.message, "✗");
  }
}

function retryPendingStudents(mergedData, blockedProgrammes, academicYearLookup) {
  try {
    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("_PendingResults");

    if (!sheet || sheet.getLastRow() <= 1) {
      logEvent("COMPLETENESS", "V", "No pending students to retry", "ℹ");
      return { resolved: [], stillPending: [] };
    }

    const cieRegnNos = new Set();
    const eseRegnNos = new Set();
    for (const key in mergedData) {
      const record = mergedData[key];
      if (record.cieMarks !== "" && record.cieMarks !== null && record.cieMarks !== undefined)
        cieRegnNos.add(record.regnNo);
      if (record.eseMarks !== "" && record.eseMarks !== null && record.eseMarks !== undefined)
        eseRegnNos.add(record.regnNo);
    }

    const lastRow = sheet.getLastRow();
    const rows    = sheet.getRange(2, 1, lastRow - 1, 11).getValues();
    const now     = getTimestamp();

    const resolved     = [];
    const stillPending = [];

    const updatedValues = rows.map(row => {
      const status = String(row[10]).trim();
      if (status === "RESOLVED") return row;

      const programme = String(row[0]).trim();
      const semester  = String(row[1]).trim();
      const batch     = String(row[2]).trim();
      const regnNo    = String(row[3]).trim();
      const reason    = String(row[6]).trim();

      if (!regnNo) return row;

      const progKey    = programme + "|" + semester + "|" + batch;
      let   isResolved = false;

      if      (reason.includes("MISSING_IN_CIE_AND_ESE"))
        isResolved = cieRegnNos.has(regnNo) && eseRegnNos.has(regnNo);
      else if (reason.includes("MISSING_IN_CIE"))
        isResolved = cieRegnNos.has(regnNo);
      else if (reason.includes("MISSING_IN_ESE"))
        isResolved = eseRegnNos.has(regnNo);
      else
        isResolved = !blockedProgrammes[progKey];

      const updated = [...row];

      if (isResolved) {
        resolved.push({ regnNo, programme, semester, batch });
        updated[8]  = now;
        updated[9]  = "RESOLVED";
        updated[10] = "RESOLVED";
      } else {
        stillPending.push({ regnNo, programme, semester, batch });
        updated[8] = now;
        updated[9] = (Number(row[9]) || 1) + 1;
      }

      return updated;
    });

    sheet.getRange(2, 1, updatedValues.length, 11).setValues(updatedValues);

    const resolvedRegnNos = new Set(resolved.map(r => r.regnNo));
    const greenIndices    = updatedValues
      .map((row, i) => (String(row[10]).trim() === "RESOLVED" && resolvedRegnNos.has(String(row[3]).trim()) ? i : -1))
      .filter(i => i !== -1);

    for (const idx of greenIndices) {
      sheet.getRange(idx + 2, 1, 1, 11).setBackground("#90EE90");
    }

    logEvent("COMPLETENESS", "V",
      "Pending retry → resolved: " + resolved.length +
      " | still pending: " + stillPending.length, "✓");

    return { resolved, stillPending };

  } catch (error) {
    logEvent("ERROR", "V", "retryPendingStudents failed: " + error.message, "✗");
    return { resolved: [], stillPending: [] };
  }
}

const _deduplicateProgrammeMap = (map) => {
  const result = {};
  const keys   = Object.keys(map);

  for (const pk of keys) {
    const [prog, sem, batch] = pk.split("|");
    // Check if a combined key already covers this split prog
    const isCoveredByCombined = prog && !prog.includes("/") && keys.some(otherPk => {
      const [otherProg, otherSem, otherBatch] = otherPk.split("|");
      return otherSem   === sem   &&
             otherBatch === batch &&
             otherProg.includes("/") &&
             otherProg.split("/").map(p => p.trim()).includes(prog);
    });
    if (!isCoveredByCombined) result[pk] = map[pk];
  }
  return result;
};

    if (!reroutedStudents || reroutedStudents.length === 0) {
      push(["No detained students this cycle"]);
    } else {
      push(["Regd No", "Name", "Programme", "Semester",
        "Original Batch", "Processing With CMF Batch"], null, true, null);
      for (const r of reroutedStudents) {
        const s = academicYearLookup ? academicYearLookup[r.regnNo] : null;
        push([
          r.regnNo,
          s ? (s.name || "—") : "—",
          r.programme,
          r.semester,
          r.fromBatch,
          r.toBatch
        ], "#EDE7F6");
      }
    }

    const totalRows = allValues.length;
    if (totalRows === 0) return;

    sheet.getRange(1, 1, totalRows, COLS).setValues(allValues);
    sheet.getRange(1, 1, totalRows, COLS).setFontWeights(allBolds);
    _applyColorRanges(sheet, allBackgrounds, allFontColors, totalRows, COLS);
    sheet.autoResizeColumns(1, COLS);
    sheet.setFrozenRows(1);

    logEvent("COMPLETENESS", "V",
      "_CompletenessReport written (" + totalRows + " rows, batched)", "✓");

  } catch (error) {
    logEvent("ERROR", "V", "_writeCompletenessReport failed: " + error.message, "✗");
  }
}

function _applyColorRanges(sheet, allBackgrounds, allFontColors, totalRows, COLS) {
  let i = 0;
  while (i < totalRows) {
    const bg = allBackgrounds[i][0];
    const fg = allFontColors[i][0];

    if (!bg && !fg) { i++; continue; }

    // Find contiguous block with same bg+fg
    let j = i + 1;
    while (j < totalRows &&
           allBackgrounds[j][0] === bg &&
           allFontColors[j][0]  === fg) {
      j++;
    }

    // j is now the first row that differs
    const rowCount = j - i;
    const rng      = sheet.getRange(i + 1, 1, rowCount, COLS);

    if (bg) rng.setBackground(bg);
    if (fg) rng.setFontColor(fg);

    i = j;
  }
}

function _loadValidationOverrides() {
  try {
    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("_ValidationOverrides");

    if (!sheet || sheet.getLastRow() <= 1) {
      Logger.log("ℹ No _ValidationOverrides sheet found — no overrides active");
      return { studentOverrides: new Map(), paperOverrides: new Map() };
    }

    const lastRow  = sheet.getLastRow();
    const lastCol  = sheet.getLastColumn();

    // ── Detect sheet format from column A header ──────────────────────────
    // New 2-col format:    col A = "Paper Code", col B = "Reason"
    // Old 3-col format:    col A = "Regd No",    col B = "Paper Code", col C = "Reason"
    // Old 2-col format:    col A = "Regd No",    col B = "Reason"
    const headerA = lastCol >= 1
      ? String(sheet.getRange(1, 1).getValue()).trim().toUpperCase()
      : "";

    // If column A header mentions "REGD" it is a legacy sheet (operator hasn't recreated it)
    const isLegacy = headerA.includes("REGD") || headerA === "REG NO";

    const studentOverrides = new Map();   // regnNo    → reason  (legacy only)
    const paperOverrides   = new Map();   // paperCode → reason  (applies to ALL students)

    if (isLegacy) {
      // ── Legacy format: col A = Regd No, col B = Paper Code (or Reason), col C = Reason ──
      // Treat any value in col A that doesn't look like a registration number as a paper code
      Logger.log("⚠ _ValidationOverrides: legacy sheet detected (col A = Regd No) — " +
        "migrating paper-code entries automatically");

      const colCount = Math.min(lastCol, 3);
      const data     = sheet.getRange(2, 1, lastRow - 1, colCount).getValues();

      for (const row of data) {
        const colA = String(row[0] || "").trim();
        const colB = String(row[1] || "").trim();
        const colC = colCount >= 3 ? String(row[2] || "").trim() : "";

        if (!colA && !colB) continue;  // blank row

        if (!colA && colB) {
          // Col A blank, col B filled → paper code in col B (3-col layout)
          paperOverrides.set(colB, colC || "No reason provided");
          continue;
        }

        if (colA) {
          // Col A filled — decide: is it a Regd No or a Paper Code?
          // Regd Nos at SSSIHL are purely numeric (e.g. 232101, 24010303006)
          const looksLikeRegnNo = /^\d+$/.test(colA);

          if (looksLikeRegnNo) {
            // Genuine student override — col B = Paper Code (ignore in new model),
            // col C = Reason. We skip student-level overrides in the new model.
            Logger.log("ℹ Legacy student override skipped (no longer supported): " + colA);
          } else {
            // Col A contains a paper code entered in the Regd No column
            // Reason is in col B (old 2-col) or col C (old 3-col)
            const reason = colC || colB || "No reason provided";
            paperOverrides.set(colA, reason);
          }
        }
      }

      Logger.log("✓ Legacy migration → paper overrides loaded: " + paperOverrides.size);

    } else {
      // ── New 2-col format: col A = Paper Code, col B = Reason ─────────────
      const colCount = Math.min(lastCol, 2);
      const data     = sheet.getRange(2, 1, lastRow - 1, colCount).getValues();

      for (const row of data) {
        const paperCode = String(row[0] || "").trim();
        const reason    = String(row[1] || "").trim();

        if (paperCode) {
          paperOverrides.set(paperCode, reason || "No reason provided");
        }
      }
    }

    Logger.log("✓ Loaded " + studentOverrides.size + " student override(s)" +
      " and " + paperOverrides.size + " paper override(s)" +
      (isLegacy ? " [legacy 2-col sheet]" : ""));

    return { studentOverrides, paperOverrides };

  } catch (error) {
    Logger.log("Error loading validation overrides: " + error.message);
    return { studentOverrides: new Map(), paperOverrides: new Map() };
  }
}

function _writeMissingStudentsPending(missingRegnNos, academicYearLookup) {
  try {
    if (Object.keys(missingRegnNos).length === 0) return;

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName("_PendingResults");

    if (!sheet) {
      sheet = ss.insertSheet("_PendingResults");
      const hdr = [
        "Programme", "Semester", "Batch", "Regd No", "Name",
        "Arrived Papers", "Blocking Reasons",
        "First Blocked On", "Last Checked On", "Runs Waited", "Status"
      ];
      sheet.getRange(1, 1, 1, hdr.length).setValues([hdr]);
      const hr = sheet.getRange(1, 1, 1, hdr.length);
      hr.setFontWeight("bold").setBackground("#000000").setFontColor("#FFFFFF");
      sheet.setFrozenRows(1);
    }

    const lastRow     = sheet.getLastRow();
    const existingMap = {};
    if (lastRow > 1) {
      sheet.getRange(2, 1, lastRow - 1, 11).getValues()
        .forEach((r, i) => {
          const rn = String(r[3]).trim();  // Regd No now at index 3
          if (rn && r[10] !== "RESOLVED") existingMap[rn] = i + 2;
        });
    }

    const now     = getTimestamp();
    const newRows = [];

    for (const regnNo in missingRegnNos) {
      const m       = missingRegnNos[regnNo];
      const student = academicYearLookup[regnNo];
      const name    = student ? student.name        : "Unknown";
      const batch   = student ? (student.batch || "") : "";

      let reason;
      if (m.missingIn.includes("CIE") && m.missingIn.includes("ESE")) {
        reason = "MISSING_IN_CIE_AND_ESE — no marks submitted in either file";
      } else if (m.missingIn.includes("CIE")) {
        reason = "MISSING_IN_CIE — CIE marks not yet submitted";
      } else {
        reason = "MISSING_IN_ESE — ESE marks not yet submitted";
      }

      if (existingMap[regnNo]) {
        const rowNum = existingMap[regnNo];
        const prev   = Number(sheet.getRange(rowNum, 10).getValue()) || 1;
        sheet.getRange(rowNum, 7).setValue(reason);   // Blocking Reasons col
        sheet.getRange(rowNum, 9).setValue(now);       // Last Checked On col
        sheet.getRange(rowNum, 10).setValue(prev + 1); // Runs Waited col
      } else {
        newRows.push([
          m.programme,
          m.semester,
          batch,
          regnNo,
          name,
          0,
          reason,
          now,
          now,
          1,
          "PENDING"
        ]);
      }
    }

    if (newRows.length > 0) {
      const startRow = sheet.getLastRow() + 1;
      sheet.getRange(startRow, 1, newRows.length, 11).setValues(newRows);

      for (let i = 0; i < newRows.length; i++) {
        const reason = newRows[i][6];
        let bg;
        if (reason.includes("CIE_AND_ESE")) bg = "#FF5252";
        else if (reason.includes("CIE"))    bg = "#FF8A65";
        else                                bg = "#FFD54F";
        sheet.getRange(startRow + i, 1, 1, 11).setBackground(bg);
      }
    }

    logEvent("COMPLETENESS", "V",
      "_PendingResults (missing students) → new: " + newRows.length +
      " | updated: " + (Object.keys(missingRegnNos).length - newRows.length), "✓");

  } catch (error) {
    logEvent("ERROR", "V", "_writeMissingStudentsPending failed: " + error.message, "✗");
  }
}

function validateCompleteness(cieRecords, eseRecords, academicYearLookup) {
  try {
    logEvent("COMPLETENESS", "V", "════ STARTING COMPLETENESS VALIDATION ════", "ℹ");
    const startTime = Date.now();

    const leftStudents = _identifyLeftStudents(cieRecords, eseRecords);
    const leftRegnNos  = new Set(Object.keys(leftStudents));
    logEvent("COMPLETENESS", "V", "Left students (L): " + leftRegnNos.size, "ℹ");

    // ── CHANGED: destructure two maps from the updated loader ─────────────
    const { studentOverrides, paperOverrides } = _loadValidationOverrides();
    logEvent("COMPLETENESS", "V",
      "Manual overrides → students: " + studentOverrides.size +
      " | papers: " + paperOverrides.size, "ℹ");

    // ── CHANGED: only whole-student overrides go into excludedRegnNos ──────
    const excludedRegnNos  = new Set([...leftRegnNos, ...studentOverrides.keys()]);

    // ── CHANGED: paper-level exclusion key set ─────────────────────────────
    const excludedPaperKeys = new Set(paperOverrides.keys());   // paperCode (all students)

    const cmfWeightageMap = readCMFWeightages();

    logEvent("COMPLETENESS", "V", "Resolving detained/gap student batches...", "ℹ");
    const { expectedMap: cmfExpectedMap } = _buildExpectedCoursesMap();
    const { reroutedStudents, ambiguousStudents } = _resolveStudentBatches(
      academicYearLookup, cmfExpectedMap
    );
    logEvent("COMPLETENESS", "V",
      "Batch rerouting → rerouted: " + reroutedStudents.length +
      " | ambiguous: " + ambiguousStudents.length, "ℹ");

    const marksArrivedRegnNos = new Set();
    for (const r of cieRecords) {
      const markStr = String(r.marks ?? "").trim().toUpperCase();
      if (markStr !== "" && markStr !== "L") marksArrivedRegnNos.add(r.regnNo);
    }
    for (const r of eseRecords) {
      const markStr = String(r.marks ?? "").trim().toUpperCase();
      if (markStr !== "" && markStr !== "L") marksArrivedRegnNos.add(r.regnNo);
    }

    for (const r of reroutedStudents) {
      const hasMarks = marksArrivedRegnNos.has(r.regnNo);
      const s        = academicYearLookup[r.regnNo];
      Logger.log((hasMarks ? "✓" : "✗") + " Rerouted student marks check: " +
        r.regnNo + " (" + (s ? s.name : "?") + ")" +
        " | " + r.programme + " | Sem " + r.semester +
        " | batch " + r.fromBatch + " → " + r.toBatch +
        " | marks arrived: " + hasMarks);
      if (!hasMarks) {
        excludedRegnNos.add(r.regnNo);
        logEvent("COMPLETENESS", "V",
          "Rerouted student excluded — no marks in fetch: " + r.regnNo, "⚠");
      }
    }

    logEvent("COMPLETENESS", "V", "Running Rule 1A: Student Count...", "ℹ");
    const rule1 = _validateStudentCount(
      cieRecords, eseRecords, academicYearLookup, excludedRegnNos
    );

    const missingRegnNos = rule1.missingRegnNos || {};
    for (const regnNo in missingRegnNos) {
      excludedRegnNos.add(regnNo);
    }

    for (const progKey in rule1.warningProgrammes) {
      const smfSet = rule1.smfSets ? rule1.smfSets[progKey] : null;
      if (!smfSet) continue;
      const remainingCount = [...smfSet].filter(r => !excludedRegnNos.has(r)).length;
      if (remainingCount === 0 && smfSet.size > 0) {
        rule1.emptyProgrammes.add(progKey);
        Logger.log("⚠ Programme promoted to empty — all students excluded: " + progKey);
      }
    }

    logEvent("COMPLETENESS", "V",
      "Rule 1A → missing students excluded: " + Object.keys(missingRegnNos).length +
      " | failed: " + Object.keys(rule1.failedProgrammes).length +
      " | empty: "  + (rule1.emptyProgrammes ? rule1.emptyProgrammes.size : 0), "ℹ");

    _writeMissingStudentsPending(missingRegnNos, academicYearLookup);

    const cieFetchPresence = new Set();
    const eseFetchPresence = new Set();

    for (const r of cieRecords) {
      if (excludedRegnNos.has(r.regnNo)) continue;
      const markStr = String(r.marks === null || r.marks === undefined ? "" : r.marks)
                        .trim().toUpperCase();
      if (markStr === "L" || markStr === "") continue;
      const s = academicYearLookup[r.regnNo];
      if (!s) continue;
      const prog  = String(s.programme || "").trim();
      const sem   = String(s.semester  || "").trim();
      const batch = String(s.batch     || "").trim();
      cieFetchPresence.add(prog + "|" + sem + "|" + batch);
      prog.split("/").map(p => p.trim()).filter(p => p).forEach(p => {
        cieFetchPresence.add(p + "|" + sem + "|" + batch);
      });
    }

    for (const r of eseRecords) {
      if (excludedRegnNos.has(r.regnNo)) continue;
      const markStr = String(r.marks === null || r.marks === undefined ? "" : r.marks)
                        .trim().toUpperCase();
      if (markStr === "L" || markStr === "") continue;
      const s = academicYearLookup[r.regnNo];
      if (!s) continue;
      const prog  = String(s.programme || "").trim();
      const sem   = String(s.semester  || "").trim();
      const batch = String(s.batch     || "").trim();
      eseFetchPresence.add(prog + "|" + sem + "|" + batch);
      prog.split("/").map(p => p.trim()).filter(p => p).forEach(p => {
        eseFetchPresence.add(p + "|" + sem + "|" + batch);
      });
    }

    logEvent("COMPLETENESS", "V",
      "Fetch presence → CIE: " + cieFetchPresence.size +
      " keys | ESE: " + eseFetchPresence.size + " keys", "ℹ");

    const builtResult = _buildExpectedCoursesMap();
    const prebuiltMap = builtResult.expectedMap;

    for (const r of reroutedStudents) {
      const progParts = r.programme.split("/").map(p => p.trim()).filter(p => p);
      const fromBatch = r.fromBatch;
      const toBatch   = r.toBatch;
      const sem       = r.semester;

      for (const prog of progParts) {
        const sourceKey = prog + "|" + sem + "|" + fromBatch;
        const targetKey = prog + "|" + sem + "|" + toBatch;
        if (!prebuiltMap[sourceKey] && prebuiltMap[targetKey]) {
          prebuiltMap[sourceKey] = prebuiltMap[targetKey];
          Logger.log("✓ CMF entry pre-injected: " + sourceKey + " → courses from " + targetKey);
        }
      }

      const combinedSourceKey = r.programme + "|" + sem + "|" + fromBatch;
      const combinedTargetKey = r.programme + "|" + sem + "|" + toBatch;
      if (!prebuiltMap[combinedSourceKey] && prebuiltMap[combinedTargetKey]) {
        prebuiltMap[combinedSourceKey] = prebuiltMap[combinedTargetKey];
      }
    }

    logEvent("COMPLETENESS", "V", "Running Rule 1B: Course Count...", "ℹ");
    // ── CHANGED: pass excludedPaperKeys ───────────────────────────────────
    const courseCount = _validateCourseCount(
      cieRecords, eseRecords, excludedRegnNos, academicYearLookup, builtResult, excludedPaperKeys
    );

    logEvent("COMPLETENESS", "V",
      "Rule 1B → hard blocked: " + Object.keys(courseCount.hardBlocks).length +
      " | warnings: "            + Object.keys(courseCount.warnings).length, "ℹ");

    for (const progKey in courseCount.warnings) {
      if (!rule1.warningProgrammes[progKey]) rule1.warningProgrammes[progKey] = {};
      rule1.warningProgrammes[progKey].courseCountWarning = courseCount.warnings[progKey];
    }

    logEvent("COMPLETENESS", "V", "Running Rule 2: Marks Quality...", "ℹ");
    // ── CHANGED: pass excludedPaperKeys ───────────────────────────────────
    const rule2 = _validateMarksQuality(
      cieRecords, eseRecords, excludedRegnNos, cmfWeightageMap, excludedPaperKeys
    );
    logEvent("COMPLETENESS", "V",
      "Rule 2 → hard blocks: " + Object.keys(rule2.flaggedRows).length +
      " | warnings: "          + Object.keys(rule2.warnings || {}).length, "ℹ");

    logEvent("COMPLETENESS", "V", "Running Rule 3: Distribution...", "ℹ");
    // ── CHANGED: pass excludedPaperKeys ───────────────────────────────────
    const rule3 = _validateDistribution(
      cieRecords, eseRecords, cmfWeightageMap, excludedRegnNos, excludedPaperKeys
    );
    logEvent("COMPLETENESS", "V",
      "Rule 3A → hard blocks: " + rule3.partA.flaggedCourses.length +
      " | warnings: "           + (rule3.partA.warnings || []).length, "ℹ");
    logEvent("COMPLETENESS", "V",
      "Rule 3B → hard blocks: " + rule3.partB.flaggedRows.length +
      " | unknown paper codes: " + rule3.unknownPaperCodes.size, "ℹ");

    const splitToCombined = {};
    for (const combinedKey of rule1.passedProgrammes) {
      const parts = combinedKey.split("|");
      const prog  = parts[0];
      const sem   = parts[1];
      const batch = parts[2] || "";
      prog.split("/").map(p => p.trim()).filter(p => p).forEach(splitProg => {
        splitToCombined[splitProg + "|" + sem + "|" + batch] = combinedKey;
      });
    }

    const allProgrammes = new Set([
      ...rule1.passedProgrammes,
      ...Object.keys(rule1.failedProgrammes),
      ...(rule1.emptyProgrammes || [])
    ]);

    for (const splitKey of Object.keys(courseCount.hardBlocks)) {
      const combined = splitToCombined[splitKey];
      if (combined) allProgrammes.add(combined);
    }

    const blockedProgrammes = {};

    for (const progKey of allProgrammes) {
      const reasons = [];
      const details = {};
      const parts   = progKey.split("|");
      const prog1b  = parts[0];
      const sem1b   = parts[1];
      const batch1b = parts[2] || "";

      const cmfHasEntry = !!(
        courseCount.expectedCoursesMap && (
          courseCount.expectedCoursesMap[progKey] ||
          prog1b.split("/").some(p =>
            courseCount.expectedCoursesMap[p + "|" + sem1b + "|" + batch1b]
          )
        )
      );
      if (!cmfHasEntry) {
        reasons.push("RULE_0_NO_CMF_BATCH_MATCH");
        details.noCMFBatchMatch = true;
      }

      if (!cieFetchPresence.has(progKey)) {
        reasons.push("NO_CIE_FETCH_DATA");
        details.noCIEFetch = true;
      }
      if (!eseFetchPresence.has(progKey)) {
        reasons.push("NO_ESE_FETCH_DATA");
        details.noESEFetch = true;
      }

      if (rule1.emptyProgrammes && rule1.emptyProgrammes.has(progKey)) {
        reasons.push("RULE_1A_ALL_STUDENTS_EXCLUDED");
        details.allStudentsExcluded = true;
      }

      if (rule1.failedProgrammes[progKey]) {
        const f = rule1.failedProgrammes[progKey];
        if (f.unexpectedInCIE && f.unexpectedInCIE.length > 0) {
          reasons.push("RULE_1A_UNEXPECTED_CIE_STUDENTS"); details.rule1a = f;
        }
        if (f.missingInCIE && f.missingInCIE.length > 0) {
          reasons.push("RULE_1A_MISSING_CIE_STUDENTS"); details.rule1a = f;
        }
        if (f.missingInESE && f.missingInESE.length > 0) {
          reasons.push("RULE_1A_MISSING_ESE_STUDENTS"); details.rule1a = f;
        }
      }

      const rule1bKeysToCheck = [
        progKey,
        ...prog1b.split("/").map(p => p.trim()).filter(p => p)
                 .map(p => p + "|" + sem1b + "|" + batch1b)
      ];
      const courseCountBlock = rule1bKeysToCheck.reduce(
        (found, k) => found || courseCount.hardBlocks[k], null
      );
      if (courseCountBlock) {
        reasons.push("RULE_1B_MISSING_COURSES");
        details.courseCount = courseCountBlock;
      }

      const rule2Hard = _rule2FlagsForProgramme(
        rule2.flaggedRows, progKey, academicYearLookup
      );
      if (rule2Hard.length > 0) {
        reasons.push("RULE_2_MARKS_QUALITY");
        details.rule2 = rule2Hard;
      }

      const partAHard = rule3.partA.flaggedCourses.filter(
        f => f.programCode === prog1b || f.programCode.includes(prog1b)
      );
      if (partAHard.length > 0) {
        reasons.push("RULE_3A_INVALID_WTG");
        details.rule3a = partAHard;
      }

      const partBHard = rule3.partB.flaggedRows.filter(f => {
        const s = academicYearLookup[f.regnNo];
        return s && (s.programme + "|" + s.semester + "|" + (s.batch || "")) === progKey;
      });
      if (partBHard.length > 0) {
        reasons.push("RULE_3B_MARKS_EXCEED_MAX");
        details.rule3b = partBHard;
      }

      if (reasons.length > 0) {
        blockedProgrammes[progKey] = { reasons, details };
      }
    }

    const eligibleProgrammes = new Set(
      [...allProgrammes].filter(p => !blockedProgrammes[p])
    );

    const duration = Math.round((Date.now() - startTime) / 1000);

    logEvent("COMPLETENESS", "V",
      "Eligible: "   + eligibleProgrammes.size +
      " | Blocked: " + Object.keys(blockedProgrammes).length +
      " | Missing: " + Object.keys(missingRegnNos).length +
      " | Empty: "   + (rule1.emptyProgrammes ? rule1.emptyProgrammes.size : 0), "✓");
    logEvent("COMPLETENESS", "V",
      "════ VALIDATION DONE in " + duration + "s ════", "✓");

    const arrivalMap = _updateProgrammeArrivalLog([...eligibleProgrammes]);

    // ── CHANGED: pass both override maps to the report ─────────────────────
    _writeCompletenessReport(
      rule1, rule2, rule3,
      blockedProgrammes, eligibleProgrammes,
      { studentOverrides, paperOverrides },
      courseCount, arrivalMap,
      reroutedStudents, academicYearLookup
    );

    // ── CHANGED: pipeline exclusion uses only student-level overrides ──────
    const pipelineExcludedRegnNos = new Set([...leftRegnNos, ...studentOverrides.keys()]);

    _writeLeftStudentsLog(leftStudents);
    _writeUnknownPaperCodesSheet(rule3.unknownPaperRecords, academicYearLookup);

    return {
      eligibleProgrammes,
      blockedProgrammes,
      leftStudents,
      missingRegnNos,
      excludedRegnNos,
      pipelineExcludedRegnNos,
      // ── CHANGED: return both override maps ─────────────────────────────
      studentOverrides,
      paperOverrides,
      excludedPaperKeys,          // Set<"regnNo|paperCode"> — used by splitEligibleAndBlocked
      overrides: studentOverrides, // backward-compat alias
      courseCount,
      unknownPaperCodes: rule3.unknownPaperCodes,
      rule1, rule2, rule3,
      duration
    };

  } catch (error) {
    logEvent("ERROR", "V", "validateCompleteness failed: " + error.message, "✗");
    throw error;
  }
}



function _buildExpectedCoursesMap() {
  try {
    const cmfFileId    = getCMFFileId();
    const cmfSheetName = getCMFSheetName();
    const cmfFile      = SpreadsheetApp.openById(cmfFileId);
    const cmfSheet     = cmfFile.getSheetByName(cmfSheetName);

    if (!cmfSheet) throw new Error("CMF sheet '" + cmfSheetName + "' not found");

    const lastRow = cmfSheet.getLastRow();
    if (lastRow < 2) throw new Error("CMF sheet is empty");

    const allData = cmfSheet.getRange(1, 1, lastRow, cmfSheet.getLastColumn()).getValues();
    const headers = allData[0];

    const colMap = {};
    for (let i = 0; i < headers.length; i++) {
      const h = String(headers[i]).trim().toUpperCase();
      if (h === "BATCH")                                    colMap.batch          = i;
      if (h === "COURSE CODE" || h === "COURSE CODE (CC)") colMap.courseCode     = i;
      if (h === "PROGRAMME CODE")                           colMap.programCode    = i;
      if (h === "PROGRAM NAME - ACRONYM")                   colMap.programCode    = i;
      if (h === "SEMESTER")                                 colMap.semester       = i;
      if (h === "CIE MAX"    || h === "CIE")               colMap.cieMax         = i;
      if (h === "ESE MAX"    || h === "ESE")                colMap.eseMax         = i;
      if (h === "COURSE CATEGORY" || h === "CATEGORY")     colMap.courseCategory = i;
    }

    const required = ["courseCode", "programCode", "semester", "cieMax", "eseMax"];
    const missing  = required.filter(k => colMap[k] === undefined);
    if (missing.length > 0) {
      Logger.log("CMF headers found: " + headers.map((h, i) => i + ":" + h).join(", "));
      throw new Error("CMF missing columns for course count: " + missing.join(", "));
    }

    const hasCategoryCol = colMap.courseCategory !== undefined;
    const hasBatchCol    = colMap.batch           !== undefined;

    if (!hasCategoryCol) Logger.log("⚠ CMF has no 'Course Category' column — all courses treated as CORE");
    if (!hasBatchCol)    Logger.log("⚠ CMF has no 'Batch' column — batch key will use empty string");

    const expectedMap      = {};
    const courseCategories = {};
    let   skipped          = 0;

    for (let i = 1; i < allData.length; i++) {
      const row         = allData[i];
      const courseCode  = String(row[colMap.courseCode]  || "").trim();
      const progRaw     = String(row[colMap.programCode] || "").trim();
      const semRaw      = String(row[colMap.semester]    || "").trim();
      const batchRaw    = hasBatchCol ? String(row[colMap.batch] || "").trim() : "";
      const categoryRaw = hasCategoryCol ? String(row[colMap.courseCategory] || "").trim() : "";

      if (!courseCode || !progRaw || !semRaw) { skipped++; continue; }

      // Skip _ASSIGN variants — credits merged into main course
      if (courseCode.toUpperCase().endsWith("_ASSIGN")) { skipped++; continue; }

      // Skip _IHA variants — IHA marks arrive in fetch under the base course code
      // (fetchCIEMarks strips _IHA at source so no _IHA codes reach the pipeline)
      if (courseCode.toUpperCase().endsWith("_IHA")) { skipped++; continue; }

      // Handle NC — "NC" means Not Counted; no marks expected in fetch for that component
      const cieMaxRaw = String(row[colMap.cieMax] === null || row[colMap.cieMax] === undefined
                          ? "" : row[colMap.cieMax]).trim();
      const eseMaxRaw = String(row[colMap.eseMax] === null || row[colMap.eseMax] === undefined
                          ? "" : row[colMap.eseMax]).trim();

      const cieIsNC = cieMaxRaw.toUpperCase() === "NC";
      const eseIsNC = eseMaxRaw.toUpperCase() === "NC";
      const cieMax  = cieIsNC ? 0 : Number(cieMaxRaw);
      const eseMax  = eseIsNC ? 0 : Number(eseMaxRaw);

      // Skip if both components have no marks expected
      if ((cieIsNC || isNaN(cieMax) || cieMax <= 0) &&
          (eseIsNC || isNaN(eseMax) || eseMax <= 0)) {
        skipped++; continue;
      }

      const category = _classifyCourse(categoryRaw);
      courseCategories[courseCode] = category;

      const allProgKeys = progRaw.split("/").map(p => p.trim()).filter(p => p);
      if (allProgKeys.length > 1) allProgKeys.unshift(progRaw); // add combined key first

      for (const prog of allProgKeys) {
        const key = prog + "|" + semRaw + "|" + batchRaw;
        if (!expectedMap[key]) {
          expectedMap[key] = {
            cieCourses:       new Set(),
            eseCourses:       new Set(),
            courseCategories: {}
          };
        }
        // Only add component to expected if it has actual marks (not NC)
        if (!cieIsNC && !isNaN(cieMax) && cieMax > 0) expectedMap[key].cieCourses.add(courseCode);
        if (!eseIsNC && !isNaN(eseMax) && eseMax > 0) expectedMap[key].eseCourses.add(courseCode);
        expectedMap[key].courseCategories[courseCode] = category;
      }
    }                        // ← closes for loop over CMF rows

    const catCounts = {};
    for (const code in courseCategories) {
      const c = courseCategories[code];
      catCounts[c] = (catCounts[c] || 0) + 1;
    }

    Logger.log("✓ _buildExpectedCoursesMap: " + Object.keys(expectedMap).length +
      " programme+semester+batch groups | " + Object.keys(courseCategories).length + " courses" +
      " | skipped: " + skipped);
    Logger.log("  Category breakdown: " + JSON.stringify(catCounts));

    return { expectedMap, courseCategories };

  } catch (error) {
    Logger.log("ERROR in _buildExpectedCoursesMap: " + error.message);
    throw error;
  }
}

function _validateCourseCount(cieRecords, eseRecords, excludedRegnNos, academicYearLookup, prebuiltExpectedMap, excludedPaperKeys) {

  const { expectedMap: expectedCoursesMap, courseCategories } =
    prebuiltExpectedMap || _buildExpectedCoursesMap();

  const paperExclusions = excludedPaperKeys instanceof Set ? excludedPaperKeys : new Set();

  const smfEnrollment        = _readSMFCategoryEnrollments(excludedRegnNos);
  const hasSMFEnrollmentData = smfEnrollment.found;
  const smfStudentCount      = _readSMFStudentCounts(excludedRegnNos, academicYearLookup);

  // ── Map A: Programme-keyed arrival maps (ALL records, no exclusion filter) ──
  //    Used for COMPLETELY ABSENT check on CORE courses
  const arrivedCIEMapAll = {};
  const arrivedESEMapAll = {};

  // ── Map B: Programme-keyed active-student maps (excluded students filtered) ──
  //    Used for partial-presence student counts on CORE courses
  const arrivedCIEMap   = {};
  const cieStudentCount = {};
  const arrivedESEMap   = {};
  const eseStudentCount = {};

  // ── Map C: Global course arrival maps (programme-independent) ──────────────
  //    Used for COMPLETELY ABSENT check and student counts on SPECIAL/ELECTIVE
  //    courses that are shared across multiple programmes.
  //    Key = courseCode → Set<regnNo that has a non-blank non-L mark>
  const globalCIEArrivals = {};
  const globalESEArrivals = {};

  // ─── Build CIE maps ────────────────────────────────────────────────────────
  for (const r of cieRecords) {
    if (paperExclusions.has(r.paperCode)) continue;

    const markStr = String(r.marks === null || r.marks === undefined ? "" : r.marks)
                      .trim().toUpperCase();
    if (markStr === "L" || markStr === "") continue;

    const student = academicYearLookup[r.regnNo];

    // Global map — any record with a valid mark, regardless of programme or exclusion
    if (!globalCIEArrivals[r.paperCode]) globalCIEArrivals[r.paperCode] = new Set();
    globalCIEArrivals[r.paperCode].add(r.regnNo);

    if (!student) continue;

    const prog  = String(student.programme || "").trim();
    const sem   = String(student.semester  || "").trim();
    const batch = String(student.batch     || "").trim();

    const allKeys = [prog + "|" + sem + "|" + batch];
    prog.split("/").map(p => p.trim()).filter(p => p).forEach(p => {
      const splitKey = p + "|" + sem + "|" + batch;
      if (!allKeys.includes(splitKey)) allKeys.push(splitKey);
    });

    // Programme-keyed unfiltered map
    for (const progKey of allKeys) {
      if (!arrivedCIEMapAll[progKey]) arrivedCIEMapAll[progKey] = new Set();
      arrivedCIEMapAll[progKey].add(r.paperCode);
    }

    // Programme-keyed active-student map
    if (excludedRegnNos.has(r.regnNo)) continue;

    for (const progKey of allKeys) {
      const courseKey = progKey + "|" + r.paperCode;
      if (!arrivedCIEMap[progKey]) arrivedCIEMap[progKey] = new Set();
      arrivedCIEMap[progKey].add(r.paperCode);
      cieStudentCount[courseKey] = (cieStudentCount[courseKey] || 0) + 1;
    }
  }

  // ─── Build ESE maps ────────────────────────────────────────────────────────
  for (const r of eseRecords) {
    if (paperExclusions.has(r.paperCode)) continue;

    const markStr = String(r.marks === null || r.marks === undefined ? "" : r.marks)
                      .trim().toUpperCase();
    if (markStr === "L" || markStr === "") continue;

    const student = academicYearLookup[r.regnNo];

    // Global map
    if (!globalESEArrivals[r.paperCode]) globalESEArrivals[r.paperCode] = new Set();
    globalESEArrivals[r.paperCode].add(r.regnNo);

    if (!student) continue;

    const prog  = String(student.programme || "").trim();
    const sem   = String(student.semester  || "").trim();
    const batch = String(student.batch     || "").trim();

    const allKeys = [prog + "|" + sem + "|" + batch];
    prog.split("/").map(p => p.trim()).filter(p => p).forEach(p => {
      const splitKey = p + "|" + sem + "|" + batch;
      if (!allKeys.includes(splitKey)) allKeys.push(splitKey);
    });

    // Programme-keyed unfiltered map
    for (const progKey of allKeys) {
      if (!arrivedESEMapAll[progKey]) arrivedESEMapAll[progKey] = new Set();
      arrivedESEMapAll[progKey].add(r.paperCode);
    }

    // Programme-keyed active-student map
    if (excludedRegnNos.has(r.regnNo)) continue;

    for (const progKey of allKeys) {
      const courseKey = progKey + "|" + r.paperCode;
      if (!arrivedESEMap[progKey]) arrivedESEMap[progKey] = new Set();
      arrivedESEMap[progKey].add(r.paperCode);
      eseStudentCount[courseKey] = (eseStudentCount[courseKey] || 0) + 1;
    }
  }

  // ─── Validation loop ───────────────────────────────────────────────────────
  const hardBlocks = {};
  const warnings   = {};

  const allCMFCIECourses = new Set();
  for (const key in expectedCoursesMap) {
    for (const c of expectedCoursesMap[key].cieCourses) allCMFCIECourses.add(c);
  }

  for (const progKey in expectedCoursesMap) {
    const entry = expectedCoursesMap[progKey];
    if (!entry || !entry.cieCourses || !entry.eseCourses) continue;

    const cieCourses = entry.cieCourses;
    const eseCourses = entry.eseCourses;
    const catMap     = entry.courseCategories || {};

    const parts = progKey.split("|");
    const prog  = parts[0];
    const sem   = parts[1];
    const batch = parts[2] || "";

    // Build variant keys (combined + split) for programme-keyed lookups
    const progVariants = [progKey];
    prog.split("/").map(p => p.trim()).filter(p => p).forEach(p => {
        const vk = p + "|" + sem + "|" + batch;
        if (!progVariants.includes(vk)) progVariants.push(vk);
    });

    // ── BATCH-WILDCARD FIX ────────────────────────────────────────────────────
    // When CMF has no Batch column (empty batch in progKey), students in SMF
    // will have a real batch year (e.g. "2024"). Their arrival records land under
    // "PROG|SEM|2024" which never matches "PROG|SEM|". Expand progVariants to
    // include every batch variant found in smfStudentCount for this prog+sem.
    if (batch === "") {
        const batchesFound = new Set();
        const ourProgs     = [prog, ...prog.split("/").map(p => p.trim()).filter(p => p)];

        for (const countKey in smfStudentCount) {
            const kp     = countKey.split("|");
            const kProg  = kp[0];
            const kSem   = kp[1];
            const kBatch = kp[2] || "";

            if (kBatch === "" || kSem !== sem) continue;

            const progMatches = ourProgs.some(op => op === kProg) ||
                kProg.split("/").map(p => p.trim()).some(kp2 => ourProgs.includes(kp2));
            if (!progMatches || batchesFound.has(kBatch)) continue;

            batchesFound.add(kBatch);
            ourProgs.forEach(op => {
                const vk = op + "|" + sem + "|" + kBatch;
                if (!progVariants.includes(vk)) progVariants.push(vk);
            });
        }

        if (batchesFound.size > 0) {
            Logger.log("ℹ Batch-wildcard expansion for " + progKey +
                " → added batch variants: " + [...batchesFound].join(", ") +
                " | total progVariants: " + progVariants.length);
        }
    }

    // Union programme-keyed arrival sets across all variant keys
    const arrivedCIEAll = new Set();
    const arrivedESEAll = new Set();
    const arrivedCIE    = new Set();
    const arrivedESE    = new Set();
    for (const vk of progVariants) {
      if (arrivedCIEMapAll[vk]) arrivedCIEMapAll[vk].forEach(c => arrivedCIEAll.add(c));
      if (arrivedESEMapAll[vk]) arrivedESEMapAll[vk].forEach(c => arrivedESEAll.add(c));
      if (arrivedCIEMap[vk])    arrivedCIEMap[vk].forEach(c => arrivedCIE.add(c));
      if (arrivedESEMap[vk])    arrivedESEMap[vk].forEach(c => arrivedESE.add(c));
    }

    let totalStudents = smfStudentCount[progKey] || 0;

    // ── BATCH-WILDCARD FIX: sum across all real-batch keys when CMF batch is empty ──
    // Only sum the exact programme code (no split variants) to avoid double-counting.
    // e.g. "BAH-E/BAHR-E" students appear under the combined key AND each split key;
    // summing splits would triple-count. "MSC-FNS" (no slash) has no split keys — safe.
    if (batch === "" && totalStudents === 0) {
        for (const countKey in smfStudentCount) {
            const kp     = countKey.split("|");
            const kProg  = kp[0];
            const kSem   = kp[1];
            const kBatch = kp[2] || "";
            if (kBatch === "" || kSem !== sem || kProg !== prog) continue;
            totalStudents += smfStudentCount[countKey];
        }
        if (totalStudents > 0) {
            Logger.log("ℹ Batch-wildcard totalStudents for " + progKey +
                " → resolved to " + totalStudents + " (summed across all batches)");
        }
    }
    // ── END BATCH-WILDCARD FIX ────────────────────────────────────────────────

    if (totalStudents === 0) continue;

    const completelyAbsent = [];
    const partiallyPresent = [];

    // ── Helper: resolve arrival status and student count for one course ───────
    // For SPECIAL courses with SMF enrollment data, check globally across all
    // programmes (cross-programme electives land under different programme keys).
    // For CORE courses (or special without enrollment data), use programme-keyed maps.
    const _resolveArrival = (courseCode, category, globalArrivals, arrivedAll, arrived, studentCountMap) => {
      const isSpecial = _isSpecialCategory(category);

      if (isSpecial && hasSMFEnrollmentData) {
        const enrolledSet = smfEnrollment.courseEnrollment[courseCode];
        if (enrolledSet && enrolledSet.size > 0) {
          // SMF enrollment known: check if any enrolled student has a mark globally
          const globalSet      = globalArrivals[courseCode];
          const anyEnrolled    = globalSet && [...enrolledSet].some(rn => globalSet.has(rn));
          const activeWithMark = globalSet
            ? [...enrolledSet].filter(rn => !excludedRegnNos.has(rn) && globalSet.has(rn)).length
            : 0;
          return { isAbsent: !anyEnrolled, studentsWithMark: activeWithMark };
        }
      }

      if (isSpecial) {
        // ── KEY FIX ──────────────────────────────────────────────────────────
        // No SMF enrollment for this elective (either no enrollment data at all,
        // or the course genuinely has no per-student enrollment listed).
        // These are cross-programme electives whose takers land under a DIFFERENT
        // batch/programme key in the fetch (e.g. elective expected for batch-2023
        // detained students but actually taken and submitted under batch-2024).
        // Using programme-keyed arrival would always show COMPLETELY ABSENT.
        // Instead: if the course arrived ANYWHERE in the fetch, treat it as present.
        // The partial-count warning can still fire if needed, but hard-blocking on
        // COMPLETELY ABSENT for a course that physically exists in the fetch is wrong.
        const globalSet      = globalArrivals[courseCode];
        const anyArrived     = !!(globalSet && globalSet.size > 0);
        // Student count: use programme-keyed count (may be 0 → partial warning)
        // or global count — whichever is higher to avoid false partial warnings
        const pkCount        = studentCountMap[progKey + "|" + courseCode] || 0;
        const globalCount    = globalSet ? globalSet.size : 0;
        const studentsWithMark = Math.max(pkCount, globalCount);
        return { isAbsent: !anyArrived, studentsWithMark };
      }

      // CORE: use programme-keyed maps — correct for non-elective courses
      const isAbsent         = !arrivedAll.has(courseCode);
      const courseKey        = progKey + "|" + courseCode;
      const studentsWithMark = studentCountMap[courseKey] || 0;
      return { isAbsent, studentsWithMark };
    };

    // ── Check CIE courses ──────────────────────────────────────────────────
    for (const courseCode of cieCourses) {
      if (paperExclusions.has(courseCode)) continue;

      const category = catMap[courseCode] || courseCategories[courseCode] || "CORE";
      const { expected, usingFallback } = _expectedCount(
        courseCode, category, totalStudents,
        smfEnrollment.courseEnrollment, hasSMFEnrollmentData
      );

      const { isAbsent, studentsWithMark } = _resolveArrival(
        courseCode, category, globalCIEArrivals, arrivedCIEAll, arrivedCIE, cieStudentCount
      );

      if (isAbsent) {
        completelyAbsent.push({
          courseCode, fetchType: "CIE", category,
          studentsWithMark: 0, totalStudents: expected, usingFallback
        });
      } else if (studentsWithMark < expected) {
        if (_isSpecialCategory(category) && !usingFallback) {
          partiallyPresent.push({
            courseCode, fetchType: "CIE", category,
            studentsWithMark, totalStudents: expected,
            missingCount: expected - studentsWithMark, usingFallback: false
          });
        } else if (!_isSpecialCategory(category)) {
          partiallyPresent.push({
            courseCode, fetchType: "CIE", category,
            studentsWithMark, totalStudents: expected,
            missingCount: expected - studentsWithMark, usingFallback
          });
        }
      }
    }

    // ── Check ESE courses ──────────────────────────────────────────────────
    for (const courseCode of eseCourses) {
      if (paperExclusions.has(courseCode)) continue;

      const category = catMap[courseCode] || courseCategories[courseCode] || "CORE";
      const { expected, usingFallback } = _expectedCount(
        courseCode, category, totalStudents,
        smfEnrollment.courseEnrollment, hasSMFEnrollmentData
      );

      const { isAbsent, studentsWithMark } = _resolveArrival(
        courseCode, category, globalESEArrivals, arrivedESEAll, arrivedESE, eseStudentCount
      );

      if (isAbsent) {
        const alreadyFlagged = completelyAbsent.some(
          c => c.courseCode === courseCode && c.fetchType === "CIE"
        );
        if (!alreadyFlagged) {
          completelyAbsent.push({
            courseCode, fetchType: "ESE", category,
            studentsWithMark: 0, totalStudents: expected, usingFallback
          });
        }
      } else if (studentsWithMark < expected) {
        const alreadyFlagged = partiallyPresent.some(c => c.courseCode === courseCode);
        if (!alreadyFlagged) {
          if (_isSpecialCategory(category) && !usingFallback) {
            partiallyPresent.push({
              courseCode, fetchType: "ESE", category,
              studentsWithMark, totalStudents: expected,
              missingCount: expected - studentsWithMark, usingFallback: false
            });
          } else if (!_isSpecialCategory(category)) {
            partiallyPresent.push({
              courseCode, fetchType: "ESE", category,
              studentsWithMark, totalStudents: expected,
              missingCount: expected - studentsWithMark, usingFallback
            });
          }
        }
      }
    }

    const arrivedRealCIE = [...arrivedCIEAll].filter(c => allCMFCIECourses.has(c));
    const cmfCIEEmpty    = cieCourses.size === 0 && arrivedRealCIE.length > 0;

    if (completelyAbsent.length > 0) {
      hardBlocks[progKey] = {
        progKey, programme: prog, semester: sem, batch,
        expectedCIE: cieCourses.size,    expectedESE: eseCourses.size,
        arrivedCIE:  arrivedCIEAll.size, arrivedESE:  arrivedESEAll.size,
        completelyAbsent, partiallyPresent,
        cmfIncomplete: cmfCIEEmpty
          ? "CIE courses arrived (" + arrivedRealCIE.join(", ") +
            ") but CMF has 0 CIE courses — CMF may be incomplete for this batch"
          : null
      };
    } else if (cmfCIEEmpty) {
      Logger.log("ℹ ESE-only (cmfCIEEmpty) — not blocking: " + progKey +
        " | unexpected CIE arrivals: " + arrivedRealCIE.join(", "));
      if (partiallyPresent.length > 0) {
        warnings[progKey] = {
          progKey, programme: prog, semester: sem, batch,
          expectedCIE: cieCourses.size,    expectedESE: eseCourses.size,
          arrivedCIE:  arrivedCIEAll.size, arrivedESE:  arrivedESEAll.size,
          completelyAbsent: [], partiallyPresent
        };
      }
    } else if (partiallyPresent.length > 0) {
      warnings[progKey] = {
        progKey, programme: prog, semester: sem, batch,
        expectedCIE: cieCourses.size,    expectedESE: eseCourses.size,
        arrivedCIE:  arrivedCIEAll.size, arrivedESE:  arrivedESEAll.size,
        completelyAbsent: [], partiallyPresent
      };
    }
  }

  Logger.log("✓ _validateCourseCount → hard blocks: " + Object.keys(hardBlocks).length +
    " | warnings: " + Object.keys(warnings).length);

  return { hardBlocks, warnings, expectedCoursesMap };
}

function _expectedCount(courseCode, category, totalStudents, courseEnrollment, hasSMFEnrollmentData) {
  if (!_isSpecialCategory(category)) {
    // CORE — always whole class
    return { expected: totalStudents, usingFallback: false };
  }

  if (!hasSMFEnrollmentData) {
    // SMF has no category columns at all — graceful degradation
    return { expected: totalStudents, usingFallback: true };
  }

  const enrolledSet = courseEnrollment[courseCode];
  if (!enrolledSet || enrolledSet.size === 0) {
    // Category columns exist but this specific course has nobody listed —
    // could be an old elective no one chose, or a data gap in SMF
    Logger.log("⚠ No SMF enrollment found for special course: " + courseCode +
      " (" + category + ") — using total class count as fallback");
    return { expected: totalStudents, usingFallback: true };
  }

  return { expected: enrolledSet.size, usingFallback: false };
}

function _readSMFStudentCounts(excludedRegnNos, academicYearLookup) {
  try {
    const studentCounts = {};

    for (const regnNo in academicYearLookup) {
      if (excludedRegnNos.has(regnNo)) continue;

      const s     = academicYearLookup[regnNo];
      const prog  = String(s.programme || "").trim();
      const sem   = String(s.semester  || "").trim();
      const batch = String(s.batch     || "").trim();

      if (!prog || !sem) continue;

      const combinedKey = prog + "|" + sem + "|" + batch;
      studentCounts[combinedKey] = (studentCounts[combinedKey] || 0) + 1;

      const progCodes = prog.split("/").map(p => p.trim()).filter(p => p);
      if (progCodes.length > 1) {
        for (const splitProg of progCodes) {
          const splitKey = splitProg + "|" + sem + "|" + batch;
          studentCounts[splitKey] = (studentCounts[splitKey] || 0) + 1;
        }
      }
    }

    Logger.log("✓ SMF student counts: " + Object.keys(studentCounts).length +
      " programme+semester+batch groups (combined + split keys)");
    return studentCounts;

  } catch (error) {
    Logger.log("ERROR in _readSMFStudentCounts: " + error.message);
    throw error;
  }
}

// ════════════════════════════════════════════════════════════════════════════
// PYTHON-STYLE COMPLETION TRACKER
// Add to: AssessmentProcessor.gs
//
// Mirrors the Python processor's tracker_df completion logic exactly:
//
//   CIE Expected = Σ campus_count_for_paper  (over all CMF papers where CIE Max > 0)
//     CORE courses    → campus_count = all unique campuses in that programme+semester
//     SPECIAL courses → campus_count = campuses with ≥1 student enrolled in that course (from SMF)
//
//   CIE Actual   = unique (campus, courseCode) pairs in the CIE fetch where mark is present
//     NOTE: CIE Actual uses campus-course pairs (not just unique course codes) to match
//           the dimensional unit of CIE Expected. This is the correct reading of the
//           Python's intent for multi-campus programmes.
//
//   ESE Expected = count of CMF courses where ESE Max > 0 (course-level, not campus-split)
//   ESE Actual   = unique course codes in ESE fetch where mark is present
//
//   Complete = CIE Expected == CIE Actual AND ESE Expected == ESE Actual
// ════════════════════════════════════════════════════════════════════════════


// ════════════════════════════════════════════════════════════════════════════
// STEP A: BUILD CAMPUS SETS PER PROGRAMME+SEMESTER
// Source: academicYearLookup (in-memory, already loaded)
// Mirrors Python: unique_campuses = smf_filtered['Campus'].unique()
// ════════════════════════════════════════════════════════════════════════════

/**
 * Derives { "prog|sem" → Set<campus> } directly from the in-memory
 * academicYearLookup so we avoid a second SMF file read.
 * Stores both the combined key ("BAH-E/BAHR-E|I") and each split key
 * ("BAH-E|I", "BAHR-E|I") so lookups work regardless of how the caller
 * formed the progKey.
 *
 * @param  {Object} academicYearLookup  — { regnNo → { programme, semester, campus, name } }
 * @param  {Set}    excludedRegnNos
 * @return {Object} campusSetByProgKey
 */
function _buildCampusSetByProgKey(academicYearLookup, excludedRegnNos) {
  const campusSetByProgKey = {};

  for (const regnNo in academicYearLookup) {
    if (excludedRegnNos.has(regnNo)) continue;

    const s = academicYearLookup[regnNo];
    if (!s || !s.campus || !s.programme || !s.semester) continue;

    const campus = String(s.campus).trim();
    const batch  = String(s.batch || "").trim();
    if (!campus) continue;

    const addCampus = key => {
      if (!campusSetByProgKey[key]) campusSetByProgKey[key] = new Set();
      campusSetByProgKey[key].add(campus);
    };

    addCampus(s.programme + "|" + s.semester + "|" + batch);

    const parts = s.programme.split("/").map(p => p.trim()).filter(p => p);
    if (parts.length > 1) parts.forEach(p => addCampus(p + "|" + s.semester + "|" + batch));
  }

  Logger.log("✓ _buildCampusSetByProgKey: " + Object.keys(campusSetByProgKey).length + " keys");
  return campusSetByProgKey;
}


// ════════════════════════════════════════════════════════════════════════════
// STEP B: BUILD SPECIAL-COURSE → CAMPUS MAP
// Source: SMF category enrollment columns (the same data _readSMFCategoryEnrollments reads)
// Mirrors Python: smf_campus_rows with paper code in paper_columns
// ════════════════════════════════════════════════════════════════════════════

/**
 * For each special-category course code, returns the Set of campuses
 * that have at least one enrolled student.
 * Derived by joining _readSMFCategoryEnrollments (courseCode → Set<regnNo>)
 * with academicYearLookup (regnNo → campus).
 *
 * @param  {Set}    excludedRegnNos
 * @param  {Object} academicYearLookup
 * @return {Object} courseCodeToCampuses: { courseCode → Set<campus> }
 */
function _buildSpecialCourseCampusMap(excludedRegnNos, academicYearLookup) {
  const smfEnrollment = _readSMFCategoryEnrollments(excludedRegnNos);

  if (!smfEnrollment.found) {
    Logger.log("ℹ _buildSpecialCourseCampusMap: no SMF category columns — returning empty map");
    return { courseCodeToCampuses: {}, hasSMFData: false };
  }

  const courseCodeToCampuses = {};

  for (const courseCode in smfEnrollment.courseEnrollment) {
    const enrolledRegnNos = smfEnrollment.courseEnrollment[courseCode];
    courseCodeToCampuses[courseCode] = new Set();

    for (const regnNo of enrolledRegnNos) {
      if (excludedRegnNos.has(regnNo)) continue;
      const s = academicYearLookup[regnNo];
      if (s && s.campus) courseCodeToCampuses[courseCode].add(String(s.campus).trim());
    }
  }

  Logger.log("✓ _buildSpecialCourseCampusMap: " + Object.keys(courseCodeToCampuses).length +
    " special courses mapped to campus sets");
  return { courseCodeToCampuses, hasSMFData: true };
}


// ════════════════════════════════════════════════════════════════════════════
// STEP C: BUILD CIE ACTUAL MAP
// Mirrors Python: combined_df['Course Code_CIE'].nunique() — but counted as
// (campus, courseCode) pairs so dimensions match CIE Expected
// ════════════════════════════════════════════════════════════════════════════

/**
 * Builds a map of unique (campus, courseCode) pairs that are present in the
 * CIE fetch, per programme+semester key.
 *
 * Python's combined_df.nunique() counts unique course codes.
 * For multi-campus programmes the Python effectively counts per individual
 * programme code (one campus each). We replicate that by counting
 * campus-course pairs — same result, correct dimension.
 *
 * @param  {Array}  cieRecords
 * @param  {Object} academicYearLookup
 * @param  {Set}    excludedRegnNos
 * @return {Object} { "prog|sem" → Set<"campus|courseCode"> }
 */
function _buildCIEActualMap(cieRecords, academicYearLookup, excludedRegnNos) {
  const cieActualMap = {};

  for (const r of cieRecords) {
    if (excludedRegnNos.has(r.regnNo)) continue;

    const markStr = String(r.marks === null || r.marks === undefined ? "" : r.marks)
                      .trim().toUpperCase();
    if (markStr === "" || markStr === "L") continue;

    const student = academicYearLookup[r.regnNo];
    if (!student || !student.campus) continue;

    const campus  = String(student.campus).trim();
    const batch   = String(student.batch || "").trim();
    const pairKey = campus + "|" + r.paperCode;

    const addPair = key => {
      if (!cieActualMap[key]) cieActualMap[key] = new Set();
      cieActualMap[key].add(pairKey);
    };

    addPair(student.programme + "|" + student.semester + "|" + batch);

    student.programme.split("/").map(p => p.trim()).filter(p => p)
      .forEach(p => addPair(p + "|" + student.semester + "|" + batch));
  }

  return cieActualMap;
}



// ════════════════════════════════════════════════════════════════════════════
// STEP D: BUILD ESE ACTUAL MAP
// Mirrors Python: combined_df['Course Code_ESE'].nunique()
// ESE is not campus-split in the Python — unique course codes only
// ════════════════════════════════════════════════════════════════════════════

/**
 * @param  {Array}  eseRecords
 * @param  {Object} academicYearLookup
 * @param  {Set}    excludedRegnNos
 * @return {Object} { "prog|sem" → Set<courseCode> }
 */
function _buildESEActualMap(eseRecords, academicYearLookup, excludedRegnNos) {
  const eseActualMap = {};

  for (const r of eseRecords) {
    if (excludedRegnNos.has(r.regnNo)) continue;

    const markStr = String(r.marks === null || r.marks === undefined ? "" : r.marks)
                      .trim().toUpperCase();
    if (markStr === "" || markStr === "L") continue;

    const student = academicYearLookup[r.regnNo];
    if (!student) continue;

    const batch = String(student.batch || "").trim();

    const addCode = key => {
      if (!eseActualMap[key]) eseActualMap[key] = new Set();
      eseActualMap[key].add(r.paperCode);
    };

    addCode(student.programme + "|" + student.semester + "|" + batch);
    student.programme.split("/").map(p => p.trim()).filter(p => p)
      .forEach(p => addCode(p + "|" + student.semester + "|" + batch));
  }

  return eseActualMap;
}


// ════════════════════════════════════════════════════════════════════════════
// MAIN: BUILD COMPLETION TRACKER
// ════════════════════════════════════════════════════════════════════════════

/**
 * Mirrors Python processor's tracker_df logic.
 *
 * Call site (in validateCompleteness, after excludedRegnNos is finalised):
 *
 *   const completionTracker = _buildCompletionTracker(
 *     cieRecords, eseRecords, academicYearLookup, excludedRegnNos
 *   );
 *
 *   // Use tracker to block incomplete programmes
 *   for (const progKey in completionTracker) {
 *     if (completionTracker[progKey].status === "Incomplete") {
 *       if (!blockedProgrammes[progKey]) blockedProgrammes[progKey] = { reasons: [], details: {} };
 *       blockedProgrammes[progKey].reasons.push("PYTHON_STYLE_INCOMPLETE");
 *       blockedProgrammes[progKey].details.completionTracker = completionTracker[progKey];
 *     }
 *   }
 *
 * @param  {Array}  cieRecords
 * @param  {Array}  eseRecords
 * @param  {Object} academicYearLookup   — { regnNo → { programme, semester, campus, name } }
 * @param  {Set}    excludedRegnNos      — left + overridden + missing students
 * @return {Object} tracker: { "prog|sem" → { cieExpected, cieActual, eseExpected, eseActual,
 *                                             status, cieGap, eseGap, totalCampuses } }
 */
function _buildCompletionTracker(cieRecords, eseRecords, academicYearLookup, excludedRegnNos) {
  const startTime = Date.now();
  logEvent("COMPLETION", "V", "Building Python-style completion tracker...", "ℹ");

  // ── A: CMF expected courses ──────────────────────────────────────────────
  const { expectedMap, courseCategories } = _buildExpectedCoursesMap();

  // ── B: Campus sets per programme+semester ───────────────────────────────
  const campusSetByProgKey = _buildCampusSetByProgKey(academicYearLookup, excludedRegnNos);

  // ── C: Special-course campus enrollment ─────────────────────────────────
  const { courseCodeToCampuses, hasSMFData } =
    _buildSpecialCourseCampusMap(excludedRegnNos, academicYearLookup);

  // ── D: CIE and ESE Actual maps ──────────────────────────────────────────
  const cieActualMap = _buildCIEActualMap(cieRecords, academicYearLookup, excludedRegnNos);
  const eseActualMap = _buildESEActualMap(eseRecords, academicYearLookup, excludedRegnNos);

  // ── E: Build tracker per programme+semester ──────────────────────────────
  const tracker       = {};
  let   completeCount = 0;

  for (const progKey in expectedMap) {
    const entry = expectedMap[progKey];
    if (!entry || !entry.cieCourses || !entry.eseCourses) continue;

    const [prog, sem]  = progKey.split("|");
    const progVariants = new Set([prog, ...prog.split("/").map(p => p.trim()).filter(p => p)]);

    // ── Resolve total campuses for this prog+sem ─────────────────────────
    // Mirrors Python: unique_campuses = smf_filtered['Campus'].unique()
    const allCampuses = new Set();
    for (const pv of progVariants) {
      const cs = campusSetByProgKey[pv + "|" + sem];
      if (cs) cs.forEach(c => allCampuses.add(c));
    }
    const totalCampuses = allCampuses.size || 1;  // guard: never divide by zero

    // ── CIE Expected ──────────────────────────────────────────────────────
    // Mirrors Python: for paper in papers_for_prog_sem → campus_count_for_paper
    let cieExpected   = 0;
    const cieGapItems = [];  // which (course, campus) pairs are absent

    for (const courseCode of entry.cieCourses) {
      const category = (entry.courseCategories || {})[courseCode]
                       || courseCategories[courseCode]
                       || "CORE";

      let campusCount;

      if (_isSpecialCategory(category) && hasSMFData) {
        // SPECIAL: only campuses with enrolled students
        const campusSet = courseCodeToCampuses[courseCode];
        if (!campusSet || campusSet.size === 0) {
          // SMF columns exist but nobody is enrolled → fallback to total campuses
          campusCount = totalCampuses;
          Logger.log("⚠ No SMF enrollment found for special course: " + courseCode +
            " (" + category + ") in " + progKey + " → using " + totalCampuses + " campuses");
        } else {
          campusCount = campusSet.size;
        }
      } else {
        // CORE (or no SMF category data): every campus takes this course
        // Mirrors Python: all campuses in smf_filtered → campus_count = 1 per campus
        campusCount = totalCampuses;
      }

      cieExpected += campusCount;
      cieGapItems.push({ courseCode, category, campusCount });
    }

    // ── ESE Expected ──────────────────────────────────────────────────────
    // Mirrors Python: pmf_prog_year[pmf_prog_year['ESE Max'] > 0].shape[0]
    const eseExpected = entry.eseCourses.size;

    // ── CIE Actual ────────────────────────────────────────────────────────
    // Unique (campus, courseCode) pairs submitted — matches CIE Expected dimension
    const cieActualSet = new Set();
    for (const pv of progVariants) {
      const ck = pv + "|" + sem;
      if (cieActualMap[ck]) cieActualMap[ck].forEach(pair => cieActualSet.add(pair));
    }
    const cieActual = cieActualSet.size;

    // ── ESE Actual ────────────────────────────────────────────────────────
    // Unique course codes submitted — matches ESE Expected dimension
    const eseActualSet = new Set();
    for (const pv of progVariants) {
      const ck = pv + "|" + sem;
      if (eseActualMap[ck]) eseActualMap[ck].forEach(code => eseActualSet.add(code));
    }
    const eseActual = eseActualSet.size;

    // ── Completion status ────────────────────────────────────────────────
    const cieComplete = cieExpected === cieActual;
    const eseComplete = eseExpected === eseActual;
    const status      = (cieComplete && eseComplete) ? "Complete" : "Incomplete";

    if (status === "Complete") completeCount++;

    tracker[progKey] = {
      programme:      prog,
      semester:       sem,
      totalCampuses,
      cieExpected,
      cieActual,
      cieGap:         cieExpected - cieActual,
      cieComplete,
      eseExpected,
      eseActual,
      eseGap:         eseExpected - eseActual,
      eseComplete,
      status,
      cieGapItems     // diagnostic: which course-campus combos are missing
    };
  }

  const duration = Math.round((Date.now() - startTime) / 1000);
  logEvent("COMPLETION", "V",
    "Tracker complete in " + duration + "s → Complete: " + completeCount +
    " | Incomplete: " + (Object.keys(tracker).length - completeCount) +
    " | Total: " + Object.keys(tracker).length, "✓");

  return tracker;
}


/**
 * Parks unknown paper code records into _UnknownPaperCodes sheet.
 * Both CIE and ESE marks are merged into a single row per regnNo+paperCode.
 * Existing entries are updated (Last Seen, Runs count).
 * New entries are appended.
 *
 * @param {Array}  unknownPaperRecords  — from _validateDistribution
 * @param {Object} academicYearLookup   — { regnNo → { programme, semester, name } }
 */
function _writeUnknownPaperCodesSheet(unknownPaperRecords, academicYearLookup) {
  try {
    if (!unknownPaperRecords || unknownPaperRecords.length === 0) {
      logEvent("COMPLETENESS", "V", "No unknown paper code records to park", "ℹ");
      return;
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName("_UnknownPaperCodes");

    if (!sheet) {
      sheet = ss.insertSheet("_UnknownPaperCodes");

      const headers = [
        "Regd No", "Paper Code", "Programme", "Semester", "Name",
        "CIE Marks", "ESE Marks",
        "First Seen On", "Last Seen On", "Runs", "Status"
      ];
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);

      const hr = sheet.getRange(1, 1, 1, headers.length);
      hr.setFontWeight("bold")
        .setBackground("#37474F")
        .setFontColor("#FFFFFF");

      sheet.setColumnWidth(1,  140);  // Regd No
      sheet.setColumnWidth(2,  200);  // Paper Code
      sheet.setColumnWidth(3,  160);  // Programme
      sheet.setColumnWidth(4,   80);  // Semester
      sheet.setColumnWidth(5,  200);  // Name
      sheet.setColumnWidth(6,   90);  // CIE Marks
      sheet.setColumnWidth(7,   90);  // ESE Marks
      sheet.setColumnWidth(8,  160);  // First Seen On
      sheet.setColumnWidth(9,  160);  // Last Seen On
      sheet.setColumnWidth(10,  60);  // Runs
      sheet.setColumnWidth(11, 100);  // Status
      sheet.setFrozenRows(1);
    }

    // ── Step 1: Merge CIE + ESE records into one entry per regnNo+paperCode ─
    // unknownPaperRecords has separate entries for CIE and ESE passes —
    // collapse them into a single row keyed on regnNo|paperCode
    const mergedUnknown = {};  // "regnNo|paperCode" → { regnNo, paperCode, cieMarks, eseMarks }

    for (const rec of unknownPaperRecords) {
      const key = rec.regnNo + "|" + rec.paperCode;
      if (!mergedUnknown[key]) {
        mergedUnknown[key] = {
          regnNo:    rec.regnNo,
          paperCode: rec.paperCode,
          cieMarks:  "",
          eseMarks:  ""
        };
      }
      if (rec.markType === "CIE") mergedUnknown[key].cieMarks = rec.submittedMark;
      if (rec.markType === "ESE") mergedUnknown[key].eseMarks = rec.submittedMark;
    }

    // ── Step 2: Load existing rows to detect updates vs new inserts ──────────
    const lastRow    = sheet.getLastRow();
    const existingRows = lastRow > 1
      ? sheet.getRange(2, 1, lastRow - 1, 11).getValues()
      : [];

    // existingMap: "regnNo|paperCode" → index in existingRows (0-based)
    const existingMap = {};
    existingRows.forEach((row, i) => {
      const key = String(row[0]).trim() + "|" + String(row[1]).trim();
      if (key !== "|" && row[10] !== "RESOLVED") existingMap[key] = i;
    });

    const now     = getTimestamp();
    const newRows = [];

    // ── Step 3: Decide update vs new insert ──────────────────────────────────
    for (const key in mergedUnknown) {
      const rec     = mergedUnknown[key];
      const student = academicYearLookup[rec.regnNo];

      const programme = student ? String(student.programme || "").trim() : "UNKNOWN";
      const semester  = student ? String(student.semester  || "").trim() : "UNKNOWN";
      const name      = student ? String(student.name      || "").trim() : "UNKNOWN";

      if (existingMap[key] !== undefined) {
        // Update CIE/ESE marks, Last Seen, Runs count in memory
        const idx = existingMap[key];
        existingRows[idx][5]  = rec.cieMarks !== "" ? rec.cieMarks : existingRows[idx][5];
        existingRows[idx][6]  = rec.eseMarks !== "" ? rec.eseMarks : existingRows[idx][6];
        existingRows[idx][8]  = now;                                  // Last Seen On
        existingRows[idx][9]  = (Number(existingRows[idx][9]) || 1) + 1;  // Runs
      } else {
        newRows.push([
          rec.regnNo,
          rec.paperCode,
          programme,
          semester,
          name,
          rec.cieMarks,   // CIE Marks — "" if only ESE was found
          rec.eseMarks,   // ESE Marks — "" if only CIE was found
          now,            // First Seen On
          now,            // Last Seen On
          1,              // Runs
          "PARKED"
        ]);
      }
    }

    // ── Step 4: Write updated existing rows back in one batch ─────────────────
    if (existingRows.length > 0) {
      sheet.getRange(2, 1, existingRows.length, 11).setValues(existingRows);
    }

    // ── Step 5: Append new rows in one batch ──────────────────────────────────
    if (newRows.length > 0) {
      const startRow = sheet.getLastRow() + 1;
      sheet.getRange(startRow, 1, newRows.length, 11).setValues(newRows);

      // Colour-code by which marks are present
      for (let i = 0; i < newRows.length; i++) {
        const hasCIE = newRows[i][5] !== "";
        const hasESE = newRows[i][6] !== "";
        let   bg;

        if (hasCIE && hasESE) bg = "#FFF9C4";  // Yellow  — both marks present
        else if (hasCIE)      bg = "#FFE0B2";  // Orange  — CIE only
        else if (hasESE)      bg = "#F3E5F5";  // Purple  — ESE only
        else                  bg = "#FFCDD2";  // Red     — no marks at all (edge case)

        sheet.getRange(startRow + i, 1, 1, 11).setBackground(bg);
      }
    }

    logEvent("COMPLETENESS", "V",
      "_UnknownPaperCodes → new: " + newRows.length +
      " | updated: " + (Object.keys(mergedUnknown).length - newRows.length) +
      " | total unique regnNo+paperCode: " + Object.keys(mergedUnknown).length, "✓");

  } catch (error) {
    logEvent("ERROR", "V", "_writeUnknownPaperCodesSheet failed: " + error.message, "✗");
  }
}

      sheet.setColumnWidth(4,   80);
      sheet.setColumnWidth(5,   90);
      sheet.setColumnWidth(6,   90);
      sheet.setColumnWidth(7,  160);
      sheet.setColumnWidth(8,  160);
      sheet.setColumnWidth(9,   60);
      sheet.setColumnWidth(10, 100);
      sheet.setColumnWidth(11, 300);
      sheet.setFrozenRows(1);
    }

    // ── Step 1: Flatten into one row per regnNo+paperCode ────────────────────
    const flatRows = {};

    for (const regnNo in notInSMFData) {
      const entry = notInSMFData[regnNo];

      for (const record of entry.records) {
        const key = regnNo + "|" + record.paperCode;

        if (!flatRows[key]) {
          flatRows[key] = {
            regnNo,
            paperCode:  record.paperCode,
            programme:  record.programme || "UNKNOWN",
            semester:   record.semester  || "UNKNOWN",
            cieMarks:   record.cieMarks  !== undefined ? record.cieMarks : "",
            eseMarks:   record.eseMarks  !== undefined ? record.eseMarks : ""
          };
        } else {
          if (flatRows[key].cieMarks === "" && record.cieMarks !== "")
            flatRows[key].cieMarks = record.cieMarks;
          if (flatRows[key].eseMarks === "" && record.eseMarks !== "")
            flatRows[key].eseMarks = record.eseMarks;
        }
      }
    }

    // ── Step 2: Load existing rows ───────────────────────────────────────────
    const lastRow      = sheet.getLastRow();
    const existingRows = lastRow > 1
      ? sheet.getRange(2, 1, lastRow - 1, 11).getValues()
      : [];

    const existingMap = {};
    existingRows.forEach((row, i) => {
      const key = String(row[0]).trim() + "|" + String(row[1]).trim();
      if (key !== "|" && String(row[9]).trim() !== "RESOLVED") {
        existingMap[key] = i;
      }
    });

    const now     = getTimestamp();
    const newRows = [];

    // ── Step 3: Update existing or queue new ─────────────────────────────────
    for (const key in flatRows) {
      const r = flatRows[key];

      if (existingMap[key] !== undefined) {
        const idx = existingMap[key];

        if (r.cieMarks !== "") existingRows[idx][4] = r.cieMarks;
        if (r.eseMarks !== "") existingRows[idx][5] = r.eseMarks;
        existingRows[idx][7] = now;
        existingRows[idx][8] = (Number(existingRows[idx][8]) || 1) + 1;

      } else {
        const hasCIE     = r.cieMarks !== "" && r.cieMarks !== null;
        const hasESE     = r.eseMarks !== "" && r.eseMarks !== null;
        const bothAbsent = !hasCIE && !hasESE;

        const remarks = bothAbsent
          ? "No marks in either CIE or ESE fetch — verify regnNo"
          : (!hasCIE
              ? "CIE marks absent in fetch"
              : (!hasESE ? "ESE marks absent in fetch" : ""));

        newRows.push([
          r.regnNo,
          r.paperCode,
          r.programme,
          r.semester,
          r.cieMarks,
          r.eseMarks,
          now,        // First Seen On
          now,        // Last Seen On
          1,          // Runs
          "PARKED",
          remarks
        ]);
      }
    }

    // ── Step 4: Batch write updated existing rows ─────────────────────────────
    if (existingRows.length > 0) {
      sheet.getRange(2, 1, existingRows.length, 11).setValues(existingRows);
    }

    // ── Step 5: Batch append new rows with colour coding ─────────────────────
    if (newRows.length > 0) {
      const startRow = sheet.getLastRow() + 1;
      sheet.getRange(startRow, 1, newRows.length, 11).setValues(newRows);

      for (let i = 0; i < newRows.length; i++) {
        const hasCIE = newRows[i][4] !== "" && newRows[i][4] !== null;
        const hasESE = newRows[i][5] !== "" && newRows[i][5] !== null;
        let   bg;

        if      (hasCIE && hasESE)  bg = "#FFF9C4";  // Yellow  — both present
        else if (hasCIE)            bg = "#FFE0B2";  // Orange  — CIE only
        else if (hasESE)            bg = "#F3E5F5";  // Purple  — ESE only
        else                        bg = "#FFCDD2";  // Red     — no marks at all

        sheet.getRange(startRow + i, 1, 1, 11).setBackground(bg);
      }
    }

    logEvent("COMPLETENESS", "V",
      "_NotInSMFStudents → new: "          + newRows.length +
      " | updated: "                       + (Object.keys(flatRows).length - newRows.length) +
      " | total unique regnNo+paperCode: " + Object.keys(flatRows).length, "✓");

  } catch (error) {
    logEvent("ERROR", "V", "_writeNotInSMFSheet failed: " + error.message, "✗");
  }
}

/**
 * Resolves detained/gap students whose batch has no CMF match.
 * If exactly one CMF batch exists for their programme+semester,
 * reroutes them to that batch (modifies academicYearLookup in place).
 * Original batch preserved as originalBatch for audit.
 *
 * @param  {Object} academicYearLookup  — modified in place
 * @param  {Object} expectedCoursesMap  — from _buildExpectedCoursesMap()
 * @return {Array}  reroutedStudents    — for logging
 */
function _resolveStudentBatches(academicYearLookup, expectedCoursesMap) {

  // ── Build: "splitProg|sem" → Set<batch> from CMF expected map ────────────
  const cmfBatchesByProgSem = {};

  for (const key in expectedCoursesMap) {
    const parts = key.split("|");
    const prog  = parts[0];
    const sem   = parts[1] || "";
    const batch = parts[2] || "";

    if (!batch) continue;  // skip empty batch entries

    const psKey = prog + "|" + sem;
    if (!cmfBatchesByProgSem[psKey]) cmfBatchesByProgSem[psKey] = new Set();
    cmfBatchesByProgSem[psKey].add(batch);
  }

  const reroutedStudents  = [];
  const ambiguousStudents = [];

  for (const regnNo in academicYearLookup) {
    const s     = academicYearLookup[regnNo];
    const batch = String(s.batch     || "").trim();
    const sem   = String(s.semester  || "").trim();

    // Split combined programme codes for CMF key lookup
    // e.g. "BAH-E/BAHR-E" → ["BAH-E", "BAHR-E"]
    const progParts = String(s.programme || "")
      .split("/").map(p => p.trim()).filter(p => p);

    // ── Check if this student's exact batch has a CMF entry ──────────────
    const hasExactMatch = progParts.some(p =>
      expectedCoursesMap[p + "|" + sem + "|" + batch]
    );

    if (hasExactMatch) continue;  // correct batch — no action needed

    // ── Collect all available CMF batches for this programme+semester ─────
    const availableBatches = new Set();
    for (const p of progParts) {
      const batches = cmfBatchesByProgSem[p + "|" + sem];
      if (batches) batches.forEach(b => availableBatches.add(b));
    }

    if (availableBatches.size === 1) {
      // Exactly one active CMF batch — safe to reroute automatically
      const targetBatch = [...availableBatches][0];

       s.isDetained  = true;        // flag for splitEligibleAndBlocked
       s.targetBatch = targetBatch; // for report labelling

      reroutedStudents.push({
          regnNo,
          programme: s.programme,
          semester:  sem,
          fromBatch: batch,
          toBatch:   targetBatch
        });

    } else if (availableBatches.size > 1) {
      // Multiple CMF batches — ambiguous, cannot auto-resolve
      // Operator must add student to _ValidationOverrides manually
      ambiguousStudents.push({
        regnNo,
        programme:       s.programme,
        semester:        sem,
        studentBatch:    batch,
        availableBatches: [...availableBatches].join(", ")
      });
      Logger.log("⚠ Ambiguous batch for " + regnNo +
        " (" + s.programme + " | " + sem + ")" +
        " — student batch: " + batch +
        " — CMF batches available: " + [...availableBatches].join(", ") +
        " — add to _ValidationOverrides to process manually");
    }
    // size === 0 → no CMF at all for this prog+sem → Rule 0 will block (correct)
  }

  // ── Log summary ───────────────────────────────────────────────────────────
  if (reroutedStudents.length > 0) {
    Logger.log("✓ Batch rerouting: " + reroutedStudents.length +
      " detained/gap student(s) rerouted to active batch:");
    for (const r of reroutedStudents) {
      Logger.log("  " + r.regnNo + " | " + r.programme + " | " + r.semester +
        " | " + r.fromBatch + " → " + r.toBatch);
    }
  }

  if (ambiguousStudents.length > 0) {
    Logger.log("⚠ Ambiguous batch (manual override required): " +
      ambiguousStudents.length + " student(s):");
    for (const a of ambiguousStudents) {
      Logger.log("  " + a.regnNo + " | " + a.programme + " | " + a.semester +
        " | student batch: " + a.studentBatch +
        " | CMF batches: " + a.availableBatches);
    }
  }

  return { reroutedStudents, ambiguousStudents };
}

function runNotInSMFCheck() {
  try {
    Logger.log("═══ NOT-IN-SMF CHECK ═══\n");

    const startTime = Date.now();

    // Load lookup and fetch marks
    const academicYearLookup = _loadAcademicYearLookup();
    const cieRecords         = fetchCIEMarks("ALL");
    const eseRecords         = fetchESEMarks("ALL");

    // Build excluded set (left students only — no full validation needed)
    const leftStudents    = _identifyLeftStudents(cieRecords, eseRecords);
    const excludedRegnNos = new Set(Object.keys(leftStudents));

    // Merge CIE + ESE
    const mergedArray = mergeCIEAndESEData(cieRecords, eseRecords);
    const mergedData  = {};
    for (const record of mergedArray) {
      mergedData[record.key] = record;
    }

    // Find students in fetch but not in lookup
    const notInSMFData = {};
    for (const key in mergedData) {
      const record = mergedData[key];
      const regnNo = record.regnNo;

      if (excludedRegnNos.has(regnNo)) continue;

      const student = academicYearLookup[regnNo];
      if (!student) {
        if (!notInSMFData[regnNo]) notInSMFData[regnNo] = { regnNo, records: [] };
        notInSMFData[regnNo].records.push(record);
      }
    }

    const count    = Object.keys(notInSMFData).length;
    const duration = Math.round((Date.now() - startTime) / 1000);

    Logger.log("Not-in-SMF students found: " + count);
    Logger.log("Duration: " + duration + "s");

    if (count > 0) {
      // Delete existing sheet for a clean run
      const ss    = SpreadsheetApp.getActiveSpreadsheet();
      const sheet = ss.getSheetByName("_NotInSMFStudents");
      if (sheet) ss.deleteSheet(sheet);

      _writeNotInSMFSheet(notInSMFData);
      Logger.log("✓ _NotInSMFStudents sheet written");

      // Log each regnNo for quick review
      Logger.log("\nParked regnNos:");
      for (const regnNo in notInSMFData) {
        Logger.log("  " + regnNo +
          " — " + notInSMFData[regnNo].records.length + " record(s)" +
          " | programme: " + (notInSMFData[regnNo].records[0].programme || "UNKNOWN"));
      }
    } else {
      Logger.log("✓ All students in fetch are present in SMF — no sheet generated");
    }

    Logger.log("\n═══ DONE ═══");

  } catch (error) {
    Logger.log("ERROR in runNotInSMFCheck: " + error.message);
    Logger.log("Stack: " + error.stack);
  }
}

function _updateProgrammeArrivalLog(eligibleArray) {
  try {
    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    let   sheet = ss.getSheetByName("_ProgrammeArrivalLog");

    if (!sheet) {
      sheet = ss.insertSheet("_ProgrammeArrivalLog");
      const hdr = [
        "Programme", "Semester", "Batch",
        "First Eligible On", "Last Seen Eligible On", "Runs Eligible"
      ];
      sheet.getRange(1, 1, 1, hdr.length).setValues([hdr]);
      sheet.getRange(1, 1, 1, hdr.length)
        .setFontWeight("bold").setBackground("#1B5E20").setFontColor("#FFFFFF");
      sheet.setFrozenRows(1);
      sheet.setColumnWidth(1, 180);
      sheet.setColumnWidth(2,  80);
      sheet.setColumnWidth(3,  80);
      sheet.setColumnWidth(4, 180);
      sheet.setColumnWidth(5, 180);
      sheet.setColumnWidth(6,  90);
    }

    // Load existing entries
    const lastRow      = sheet.getLastRow();
    const existingRows = lastRow > 1
      ? sheet.getRange(2, 1, lastRow - 1, 6).getValues()
      : [];

    // existingMap: "prog|sem|batch" → row index (0-based)
    const existingMap = {};
    existingRows.forEach((row, i) => {
      const key = String(row[0]).trim() + "|" + String(row[1]).trim() + "|" + String(row[2]).trim();
      if (key !== "||") existingMap[key] = i;
    });

    const now     = getTimestamp();
    const newRows = [];

    for (const progKey of eligibleArray) {
      const [prog, sem, batch] = progKey.split("|");

      if (existingMap[progKey] !== undefined) {
        // Update last seen + runs count
        const idx = existingMap[progKey];
        existingRows[idx][4] = now;
        existingRows[idx][5] = (Number(existingRows[idx][5]) || 1) + 1;
      } else {
        // First time this programme is eligible
        newRows.push([prog, sem, batch || "", now, now, 1]);
      }
    }

    if (existingRows.length > 0) {
      sheet.getRange(2, 1, existingRows.length, 6).setValues(existingRows);
    }

    if (newRows.length > 0) {
      const startRow = sheet.getLastRow() + 1;
      sheet.getRange(startRow, 1, newRows.length, 6).setValues(newRows);
      sheet.getRange(startRow, 1, newRows.length, 6).setBackground("#C8E6C9");
    }

    // Build and return lookup map for the report
    const arrivalMap = {};

    // Re-read all rows to get the complete picture
    const allRows = sheet.getLastRow() > 1
      ? sheet.getRange(2, 1, sheet.getLastRow() - 1, 6).getValues()
      : [];

    for (const row of allRows) {
      const key = String(row[0]).trim() + "|" + String(row[1]).trim() + "|" + String(row[2]).trim();
      arrivalMap[key] = {
        firstEligibleOn:   String(row[3]).trim(),
        lastSeenEligibleOn: String(row[4]).trim(),
        runsEligible:      Number(row[5]) || 1
      };
    }

    Logger.log("✓ _ProgrammeArrivalLog → new: " + newRows.length +
      " | updated: " + Object.keys(existingMap).length);

    return arrivalMap;

  } catch (error) {
    Logger.log("ERROR in _updateProgrammeArrivalLog: " + error.message);
    return {};
  }
}

  }

  // ── Step 3: Summary comparison ─────────────────────────────────────────
  const smfSemCredits  = student.semCredits;
  const creditMismatch = smfSemCredits !== null     &&
                         smfSemCredits !== undefined &&
                         smfSemCredits > 0           &&
                         sumCredits !== smfSemCredits;

  Logger.log("\n─── CREDIT COMPARISON ─────────────────────────────────");
  Logger.log("  Sum of Credits (Consolidated Result, col O) : " + sumCredits);
  Logger.log("  SMF Sem-Credits                              : " +
    (smfSemCredits !== null && smfSemCredits !== undefined
      ? smfSemCredits
      : "NULL (column missing or blank)"));
  Logger.log("  WITHHELD triggered                           : " + creditMismatch);

  if (smfSemCredits === null || smfSemCredits === undefined) {
    Logger.log("\n  ROOT CAUSE: SMF has no Sem-Credits column, OR the column");
    Logger.log("  exists but is blank for this student. WITHHELD should NOT");
    Logger.log("  have fired — check _loadAcademicYearLookup() semCreditsColIdx.");
  } else if (!creditMismatch) {
    Logger.log("\n  ✓ Credits match — WITHHELD should NOT have fired.");
    Logger.log("    Re-run populateFinalOutput() to regenerate FinalOutput.");
  } else {
    const diff = sumCredits - smfSemCredits;
    Logger.log("\n  ✗ MISMATCH of " + diff + " credits.");

    if (nullCreditCount > 0) {
      Logger.log("\n  LIKELY CAUSE: " + nullCreditCount +
        " paper(s) have a BLANK Credits cell in Consolidated Result (col O).");
      Logger.log("  These blank cells are counting as 0, pulling sumCredits below SMF value.");
      Logger.log("  Fix: ensure the formula in col O evaluates correctly for all rows,");
      Logger.log("  or verify the Paper Code Table has credits defined for every paper.");
    }

    if (zeroCreditCount > 0) {
      Logger.log("\n  NOTE: " + zeroCreditCount +
        " paper(s) have Credits = 0. If intentional (e.g. non-credit course) this is fine.");
      Logger.log("  If not, the Paper Code Table or CMF credits entry is missing.");
    }

    if (nullCreditCount === 0 && zeroCreditCount === 0) {
      Logger.log("\n  All credits appear populated. Check:");
      Logger.log("  1. SMF Sem-Credits value (" + smfSemCredits + ") — is it correct for this student?");
      Logger.log("  2. Is this student in a non-standard load (audit/extra paper) not in SMF?");
      Logger.log("  3. Does the student have a paper in Consolidated Result that shouldn't be there?");
    }
  }

  Logger.log("\n  Papers with blank Credits (col O):");
  let blankFound = false;
  for (const row of studentRows) {
    const rawCredits = row[14];
    if (rawCredits === "" || rawCredits === null || rawCredits === undefined) {
      Logger.log("    ✗ " + String(row[1]).trim() + " — Credits blank");
      blankFound = true;
    }
  }
  if (!blankFound) Logger.log("    None — all credit cells are populated");

  Logger.log("\n═══ DONE ═══");
  flushLogs();
}

function diagnoseMSCFNSAbsent() {
  const TARGET_PROG = "MSC-FNS";
  const TARGET_SEM  = "IV";
  const SUSPECT_CODES = ["PFNS-401(A)", "PFNS-402(A)", "PFNS-403(E-ANFT1)", "PFNS-404(A)"];

  Logger.log("═══ DIAGNOSING COMPLETELY ABSENT FOR " + TARGET_PROG + " SEM " + TARGET_SEM + " ═══\n");

  const academicYearLookup = _loadAcademicYearLookup();
  const cieRecords         = fetchCIEMarks("ALL");

  // ── Step 1: What paper codes exist in the CIE fetch for MSC-FNS students? ──
  const mscFNSRegnNos = new Set();
  for (const regnNo in academicYearLookup) {
    const s = academicYearLookup[regnNo];
    if (String(s.programme || "").trim() === TARGET_PROG &&
        String(s.semester  || "").trim() === TARGET_SEM) {
      mscFNSRegnNos.add(regnNo);
    }
  }

  Logger.log("MSC-FNS Sem IV students in SMF: " + mscFNSRegnNos.size);
  Logger.log("RegnNos: " + [...mscFNSRegnNos].join(", "));
  Logger.log("");

  // ── Step 2: What paper codes arrived for these students in CIE fetch? ──
  const arrivedCodes = new Set();
  for (const r of cieRecords) {
    if (!mscFNSRegnNos.has(r.regnNo)) continue;
    const markStr = String(r.marks ?? "").trim();
    Logger.log("  CIE record → regnNo=[" + r.regnNo + "]" +
               " paperCode=[" + r.paperCode + "]" +
               " marks=[" + markStr + "]" +
               " bytes=" + [...r.paperCode].map(c => c.charCodeAt(0)).join(","));
    arrivedCodes.add(r.paperCode);
  }

  Logger.log("\nUnique paper codes arrived in CIE fetch for MSC-FNS Sem IV:");
  for (const c of arrivedCodes) {
    Logger.log("  [" + c + "] length=" + c.length);
  }

  // ── Step 3: What does CMF expect? ──
  Logger.log("\nCMF expects (suspect codes):");
  for (const expected of SUSPECT_CODES) {
    const inArrived = arrivedCodes.has(expected);
    Logger.log("  CMF=[" + expected + "] length=" + expected.length +
               " | In fetch: " + (inArrived ? "✓ YES" : "✗ NO — MISMATCH"));
    
    if (!inArrived) {
      // Find closest match
      for (const arrived of arrivedCodes) {
        const normalized_arrived  = arrived.trim().toUpperCase().replace(/\s/g, "");
        const normalized_expected = expected.trim().toUpperCase().replace(/\s/g, "");
        if (normalized_arrived === normalized_expected) {
          Logger.log("    → Likely match (whitespace difference): [" + arrived + "]");
        } else if (normalized_arrived.replace(/[()]/g, "") === normalized_expected.replace(/[()]/g, "")) {
          Logger.log("    → Likely match (parenthesis difference): [" + arrived + "]");
        }
      }
    }
  }

  Logger.log("\n═══ DONE ═══");
  flushLogs();
}
